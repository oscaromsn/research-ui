# 13. Code Organization & Style

## 13.1 Module Structure

### File Organization

**Feature-Based Structure (Recommended):**

```
src/
├── services/
│   ├── UserService.ts
│   ├── AuthService.ts

├── repositories/

#### Domain Module Template

```typescript
// domain/User.ts
import * as S from "@effect/schema/Schema"
import { Data, Effect } from "effect"

// 1. Constants
export const USER_NAME_MIN_LENGTH = 2
export const USER_NAME_MAX_LENGTH = 100
export const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]+$/

// 2. Branded types
export const UserId = S.String.pipe(S.brand("UserId"))
export type UserId = S.Schema.Type<typeof UserId>

// 3. Schema definitions
export class User extends S.Class<User>("User")({
  id: UserId,
  name: S.String.pipe(
    S.minLength(USER_NAME_MIN_LENGTH),
    S.maxLength(USER_NAME_MAX_LENGTH)
  ),
  email: S.String.pipe(S.pattern(EMAIL_PATTERN)),
  createdAt: S.DateTimeUtc,
  updatedAt: S.DateTimeUtc
}) {}

// 4. Errors
export class UserError extends Data.TaggedError("UserError")<{
  readonly message: string
  readonly cause?: unknown
}> {}

// 5. Operations
export const validateUser = (data: unknown): Effect.Effect<User, UserError> =>
  S.decodeUnknown(User)(data).pipe(
    Effect.mapError((error) => new UserError({
      message: "User validation failed",
      cause: error
    }))
  )

// 6. Helper functions
export const isValidEmail = (email: string): boolean =>
  EMAIL_PATTERN.test(email)

export const formatUserName = (user: User): string =>
  user.name.trim()
```

#### Utility Module Template

```typescript
// utils/validation.ts
import { Effect, Data } from "effect"
import * as S from "@effect/schema/Schema"

// 1. Errors
export class ValidationError extends Data.TaggedError("ValidationError")<{
  readonly field: string
  readonly message: string
}> {}

// 2. Generic validation utilities
export const validateField = <T>(
  schema: S.Schema<T>,
  field: string
) => (value: unknown): Effect.Effect<T, ValidationError> =>
  S.decodeUnknown(schema)(value).pipe(
    Effect.mapError(() => new ValidationError({
      field,
      message: `Invalid ${field}`
    }))
  )

// 3. Specific validators
export const validateEmail = validateField(
  S.String.pipe(S.pattern(/^[^\s@]+@[^\s@]+\.[^\s@]+$/)),
  "email"
)

export const validateAge = validateField(
  S.Number.pipe(S.int(), S.between(0, 150)),
  "age"
)

// 4. Composition helpers
export const validateAll = <T extends Record<string, unknown>>(
  validators: { [K in keyof T]: (value: unknown) => Effect.Effect<T[K], ValidationError> }
) => (data: unknown): Effect.Effect<T, ValidationError> =>
  Effect.gen(function* () {
    const result = {} as T
    for (const [key, validator] of Object.entries(validators)) {
      result[key as keyof T] = yield* validator((data as any)[key])
    }
    return result
  })
```

### Module Dependencies

**Import Organization:**

```typescript
// Correct order of imports
// 1. Effect core imports
import { Effect, Context, Layer, Data } from "effect"
import * as S from "@effect/schema/Schema"

// 2. Effect platform imports
import { HttpClient, HttpServer } from "@effect/platform"

// 3. Third-party imports

// 4. Internal absolute imports

// 5. Internal relative imports

import type { User, UserData } from "./types"
```

**Dependency Direction:**

```typescript
// ✅ CORRECT: Dependencies flow inward
// domain/ (no dependencies)
// ├── User.ts
// ├── Order.ts
//
// repositories/ (depends on domain)
// ├── UserRepository.ts → imports User
//
// services/ (depends on repositories + domain)
// ├── UserService.ts → imports UserRepository, User
//
// handlers/ (depends on services)
// ├── UserHandler.ts → imports UserService

// ❌ WRONG: Circular dependencies
// services/UserService.ts imports handlers/UserHandler.ts
// handlers/UserHandler.ts imports services/UserService.ts
```

**Avoiding Circular Dependencies:**

```typescript
// ❌ BAD: Circular dependency
// ServiceA.ts
import { ServiceB } from "./ServiceB"
export class ServiceA extends Context.Tag("ServiceA")<ServiceA, {
  doSomething: Effect.Effect<void>
}>() {}

// ServiceB.ts
import { ServiceA } from "./ServiceA" // Circular!
export class ServiceB extends Context.Tag("ServiceB")<ServiceB, {
  doSomethingElse: Effect.Effect<void>
}>() {}

// ✅ GOOD: Extract shared types
// types.ts
export interface ServiceAInterface {
  doSomething: Effect.Effect<void>
}

export interface ServiceBInterface {
  doSomethingElse: Effect.Effect<void>
}

// ServiceA.ts
import type { ServiceBInterface } from "./types"
export class ServiceA extends Context.Tag("ServiceA")<ServiceA, ServiceAInterface>() {}

// ServiceB.ts
import type { ServiceAInterface } from "./types"
export class ServiceB extends Context.Tag("ServiceB")<ServiceB, ServiceBInterface>() {}
```

### Smart Constructors Pattern

```typescript
// Effect-based constructor (returns Effect)
/**
 * Construct a KeyHash from a hex string.
 *
 * @example
 * import { KeyHash } from "@app/domain"
 * import { Effect } from "effect"
 *
 * const hash = "c37b1b5dc0669f1d3c61a6fddb2e8fde96be87b881c60bce8e8d542f"
 * const keyHashEffect = KeyHash.make(hash)
 * const keyHash = Effect.runSync(keyHashEffect)
 *
 * @since 2.0.0
 * @category constructors
 */
export const make = (hash: string): Effect.Effect<KeyHash, KeyHashError> =>
  Effect.gen(function* () {
    if (hash.length !== KEYHASH_HEX_LENGTH) {
      return yield* new KeyHashError({
        message: `KeyHash must be ${KEYHASH_HEX_LENGTH} characters long, got ${hash.length}`
      })
    }
    if (!Bytes.isHex(hash)) {
      return yield* new KeyHashError({
        message: "KeyHash must be a valid hex string"
      })
    }
    return new KeyHash({ hash }, { disableValidation: true })
  })

// OrThrow variant (throws instead of returning Effect)
/**
 * Construct a KeyHash from a hex string, throws on error.
 *
 * @example
 * import { KeyHash } from "@app/domain"
 *
 * const hash = "c37b1b5dc0669f1d3c61a6fddb2e8fde96be87b881c60bce8e8d542f"
 * const keyHash = KeyHash.makeOrThrow(hash)
 *
 * @since 2.0.0
 * @category constructors
 */
export const makeOrThrow = (hash: string): KeyHash => {
  if (hash.length !== KEYHASH_HEX_LENGTH) {
    throw new KeyHashError({
      message: `KeyHash must be ${KEYHASH_HEX_LENGTH} characters long, got ${hash.length}`
    })
  }
  if (!Bytes.isHex(hash)) {
    throw new KeyHashError({
      message: "KeyHash must be a valid hex string"
    })
  }
  return new KeyHash({ hash }, { disableValidation: true })
}
```

## 13.4 Code Quality Standards

### Type Safety Rules

```typescript
// ✅ CORRECT: Explicit types, no assertions
const user: User = {
  id: "123",
  name: "John Doe",
  email: "john@example.com"
}

// ✅ CORRECT: Use unknown and validate
const processData = (input: unknown): Effect.Effect<ProcessedData, ValidationError> =>

    return process(validated)
  })

const config = {
  host: "localhost",
  port: 5432,
  maxConnections: 10
} satisfies DatabaseConfig

// ❌ FORBIDDEN: Never use any or type assertions
const data: any = fetchData()  // NEVER!
const user = data as User  // NEVER!
const result = something as never  // NEVER!

### Code Clarity

```typescript
// ✅ CORRECT: Early returns for readability
const validateUser = (user: User): Effect.Effect<void, ValidationError> =>
  Effect.gen(function* () {
    if (!user.name) {
      return yield* new ValidationError({ field: "name", message: "Name is required" })
    }
    
    if (!user.email) {
      return yield* new ValidationError({ field: "email", message: "Email is required" })
    }
    
    if (!isValidEmail(user.email)) {
      return yield* new ValidationError({ field: "email", message: "Invalid email format" })
    }
    
    // Continue with processing
  })

// ❌ WRONG: Deep nesting
const validateUser = (user: User): Effect.Effect<void, ValidationError> =>
  Effect.gen(function* () {
    if (user.name) {
      if (user.email) {
        if (isValidEmail(user.email)) {
          // Processing buried deep in nesting
        } else {
          return yield* new ValidationError({ field: "email", message: "Invalid email" })
        }
      } else {
        return yield* new ValidationError({ field: "email", message: "Email required" })
      }
    } else {
      return yield* new ValidationError({ field: "name", message: "Name required" })
    }
  })

// ✅ CORRECT: Guard clauses
const processOrder = (order: Order): Effect.Effect<ProcessedOrder, OrderError> =>
  Effect.gen(function* () {
    // Guard clauses at the top
    if (order.items.length === 0) {
      return yield* new OrderError({ message: "Order must have items" })
    }
    
    if (order.total <= 0) {
      return yield* new OrderError({ message: "Order total must be positive" })
    }
    
    // Main processing logic
    const validated = yield* validateOrder(order)
    const processed = yield* applyDiscounts(validated)
    return processed
  })
```

```typescript
/**

 *
 * @example
 * ```ts

 * ```
 *
 * @example
 * ```ts
 * import { validateUser } from "@app/validation"
 * import { Effect } from "effect"
 *
 * // ❌ Invalid user data - missing email
 * const invalidUser = {
 *   name: "Jane Doe",
 *   age: 25
 * }
 *
 * const program = Effect.gen(function* () {

 *

export const validateUser = (data: unknown): Effect.Effect<User, ValidationError> =>
  S.decodeUnknown(UserSchema)(data).pipe(
    Effect.mapError((error) => new ValidationError({
      message: "User validation failed",
      cause: error
    }))
  )
```

---

**Best Practices Summary:**

1. **Module Organization**: Use feature-based structure with clear separation of concerns
2. **Naming Conventions**: Follow consistent patterns (camelCase for functions/variables, PascalCase for classes/types)
3. **Function Patterns**: Provide both Effect-based and OrThrow variants
4. **Type Safety**: Never use `any`, `as never`, or type assertions
5. **Code Clarity**: Use early returns, guard clauses, minimal nesting
6. **Documentation**: Include JSDoc with examples for all public functions
7. **Examples**: Provide complete, runnable code examples
8. **Categories**: Use consistent @category tags for organization
9. **Functional Style**: Prefer pure functions, immutability, and composition
