# 16. Advanced Patterns

This section covers sophisticated architectural patterns that can be implemented elegantly with Effect-TS, including state machines, event-driven architectures, CQRS, and sagas.

## 16.1 State Machines

State machines provide explicit modeling of system states and valid transitions, making complex business logic more maintainable and testable.

### Core State Machine Pattern

```typescript
import { Effect, Data, Schema as S, Match } from "effect"

// Define states using discriminated unions
export type OrderState =
  | { readonly _tag: "Pending"; readonly orderId: string }
  | { readonly _tag: "PaymentProcessing"; readonly orderId: string; readonly paymentId: string }
  | { readonly _tag: "Confirmed"; readonly orderId: string; readonly paymentId: string }
  | { readonly _tag: "Shipped"; readonly orderId: string; readonly trackingId: string }
  | { readonly _tag: "Delivered"; readonly orderId: string; readonly deliveredAt: Date }
  | { readonly _tag: "Cancelled"; readonly orderId: string; readonly reason: string }

// Define events (triggers for transitions)
export type OrderEvent =
  | { readonly _tag: "PaymentInitiated"; readonly paymentId: string }
  | { readonly _tag: "PaymentConfirmed" }
  | { readonly _tag: "OrderShipped"; readonly trackingId: string }
  | { readonly _tag: "OrderDelivered"; readonly deliveredAt: Date }
  | { readonly _tag: "OrderCancelled"; readonly reason: string }

// Define errors
export class InvalidTransitionError extends Data.TaggedError("InvalidTransitionError")<{
  readonly from: string
  readonly event: string
}> {}

export class StateMachineError extends Data.TaggedError("StateMachineError")<{
  readonly message: string
  readonly cause?: unknown
}> {}

// State machine transition logic
export const transition = (
  state: OrderState,
  event: OrderEvent
): Effect.Effect<OrderState, InvalidTransitionError> =>
  Match.value(state).pipe(
    Match.tag("Pending", (pending) =>
      Match.value(event).pipe(
        Match.tag("PaymentInitiated", (evt) =>
          Effect.succeed({
            _tag: "PaymentProcessing" as const,
            orderId: pending.orderId,
            paymentId: evt.paymentId
          })
        ),
        Match.tag("OrderCancelled", (evt) =>
          Effect.succeed({
            _tag: "Cancelled" as const,
            orderId: pending.orderId,
            reason: evt.reason
          })
        ),
        Match.orElse(() =>
          Effect.fail(new InvalidTransitionError({
            from: "Pending",
            event: event._tag
          }))
        )
      )
    ),
    Match.tag("PaymentProcessing", (processing) =>
      Match.value(event).pipe(
        Match.tag("PaymentConfirmed", () =>
          Effect.succeed({
            _tag: "Confirmed" as const,
            orderId: processing.orderId,
            paymentId: processing.paymentId
          })
        ),
        Match.tag("OrderCancelled", (evt) =>
          Effect.succeed({
            _tag: "Cancelled" as const,
            orderId: processing.orderId,
            reason: evt.reason
          })
        ),
        Match.orElse(() =>
          Effect.fail(new InvalidTransitionError({
            from: "PaymentProcessing",
            event: event._tag
          }))
        )
      )
    ),
    Match.tag("Confirmed", (confirmed) =>
      Match.value(event).pipe(
        Match.tag("OrderShipped", (evt) =>
          Effect.succeed({
            _tag: "Shipped" as const,
            orderId: confirmed.orderId,
            trackingId: evt.trackingId
          })
        ),
        Match.orElse(() =>
          Effect.fail(new InvalidTransitionError({
            from: "Confirmed",
            event: event._tag
          }))
        )
      )
    ),
    Match.tag("Shipped", (shipped) =>
      Match.value(event).pipe(
        Match.tag("OrderDelivered", (evt) =>
          Effect.succeed({
            _tag: "Delivered" as const,
            orderId: shipped.orderId,
            deliveredAt: evt.deliveredAt
          })
        ),
        Match.orElse(() =>
          Effect.fail(new InvalidTransitionError({
            from: "Shipped",
            event: event._tag
          }))
        )
      )
    ),
    Match.orElse((state) =>
      Effect.fail(new InvalidTransitionError({
        from: state._tag,
        event: event._tag
      }))
    )
  )

// Guards: Conditional transitions based on business rules
export const canCancelOrder = (state: OrderState): boolean =>
  Match.value(state).pipe(
    Match.tag("Pending", () => true),
    Match.tag("PaymentProcessing", () => true),
    Match.tag("Confirmed", () => true),
    Match.tag("Shipped", () => false),
    Match.tag("Delivered", () => false),
    Match.tag("Cancelled", () => false),
    Match.exhaustive
  )

// Actions: Side effects executed during transitions
export const executeAction = (
  state: OrderState,
  event: OrderEvent
): Effect.Effect<void, StateMachineError> =>
  Match.value({ state, event }).pipe(
    Match.when(
      { state: { _tag: "PaymentProcessing" }, event: { _tag: "PaymentConfirmed" } },
      () => Effect.gen(function* () {
        yield* Effect.log("Notifying customer of confirmation")
        yield* Effect.sleep("100 millis") // Simulate notification
      })
    ),
    Match.when(
      { state: { _tag: "Confirmed" }, event: { _tag: "OrderShipped" } },
      () => Effect.gen(function* () {
        yield* Effect.log("Sending shipping notification")
        yield* Effect.sleep("100 millis") // Simulate notification
      })
    ),
    Match.orElse(() => Effect.void)
  )

// State machine with side effects
export const processEvent = (
  currentState: OrderState,
  event: OrderEvent
): Effect.Effect<OrderState, InvalidTransitionError | StateMachineError> =>
  Effect.gen(function* () {
    // Execute pre-transition actions
    yield* executeAction(currentState, event)

    // Perform transition
    const newState = yield* transition(currentState, event)

    // Log transition
    yield* Effect.log("State transition", {
      from: currentState._tag,
      to: newState._tag,
      event: event._tag
    })

    return newState
  })
```

### State Machine Service

```typescript
import { Effect, Context, Layer, Ref } from "effect"

// State machine service with persistence
export class OrderStateMachine extends Context.Tag("OrderStateMachine")
  OrderStateMachine,
  {
    readonly getCurrentState: (orderId: string) => Effect.Effect<OrderState, StateMachineError>
    readonly applyEvent: (
      orderId: string,
      event: OrderEvent
    ) => Effect.Effect<OrderState, InvalidTransitionError | StateMachineError>
    readonly getHistory: (orderId: string) => Effect.Effect<ReadonlyArray<OrderEvent>, StateMachineError>
  }
>() {}

export const makeOrderStateMachine = Effect.gen(function* () {
  // In-memory state store (use database in production)
  const stateStore = yield* Ref.make<Map<string, OrderState>>(new Map())
  const historyStore = yield* Ref.make<Map<string, Array<OrderEvent>>>(new Map())

  const getCurrentState = (orderId: string): Effect.Effect<OrderState, StateMachineError> =>
    Effect.gen(function* () {
      const states = yield* Ref.get(stateStore)
      const state = states.get(orderId)

      if (!state) {
        return yield* Effect.fail(new StateMachineError({
          message: `Order ${orderId} not found`
        }))
      }

      return state
    })

  const applyEvent = (
    orderId: string,
    event: OrderEvent
  ): Effect.Effect<OrderState, InvalidTransitionError | StateMachineError> =>
    Effect.gen(function* () {
      const currentState = yield* getCurrentState(orderId)
      const newState = yield* processEvent(currentState, event)

      // Persist new state
      yield* Ref.update(stateStore, (states) =>
        new Map(states).set(orderId, newState)
      )

      // Record event in history
      yield* Ref.update(historyStore, (history) => {
        const events = history.get(orderId) || []
        return new Map(history).set(orderId, [...events, event])
      })

      return newState
    })

  const getHistory = (orderId: string): Effect.Effect<ReadonlyArray<OrderEvent>, StateMachineError> =>
    Effect.gen(function* () {
      const history = yield* Ref.get(historyStore)
      return history.get(orderId) || []
    })

  return { getCurrentState, applyEvent, getHistory } as const
})

export const OrderStateMachineLive = Layer.effect(
  OrderStateMachine,
  makeOrderStateMachine
)
```

### Testing State Machines

```typescript
import { assert, describe, it } from "@effect/vitest"
import { Effect } from "effect"

describe("OrderStateMachine", () => {
  it.effect("should transition from Pending to PaymentProcessing", () =>
    Effect.gen(function* () {
      const initialState: OrderState = {
        _tag: "Pending",
        orderId: "order-123"
      }

      const event: OrderEvent = {
        _tag: "PaymentInitiated",
        paymentId: "payment-456"
      }

      const newState = yield* transition(initialState, event)

      assert.strictEqual(newState._tag, "PaymentProcessing")
      assert.strictEqual(newState.orderId, "order-123")
      if (newState._tag === "PaymentProcessing") {
        assert.strictEqual(newState.paymentId, "payment-456")
      }
    })
  )

  it.effect("should reject invalid transitions", () =>
    Effect.gen(function* () {
      const state: OrderState = {
        _tag: "Delivered",
        orderId: "order-123",
        deliveredAt: new Date()
      }

      const event: OrderEvent = {
        _tag: "OrderCancelled",
        reason: "Customer request"
      }

      const error = yield* transition(state, event).pipe(Effect.flip)

      assert.strictEqual(error._tag, "InvalidTransitionError")
      assert.strictEqual(error.from, "Delivered")
    })
  )

  it.effect("should execute actions during transitions", () =>
    Effect.gen(function* () {
      const state: OrderState = {
        _tag: "Confirmed",
        orderId: "order-123",
        paymentId: "payment-456"
      }

      const event: OrderEvent = {
        _tag: "OrderShipped",
        trackingId: "track-789"
      }

      const newState = yield* processEvent(state, event)

      assert.strictEqual(newState._tag, "Shipped")
      if (newState._tag === "Shipped") {
        assert.strictEqual(newState.trackingId, "track-789")
      }
    })
  )
})
```

### Event Bus Service

```typescript
import { Effect, Context, Layer, Queue, Ref, Stream } from "effect"

export class EventBusError extends Data.TaggedError("EventBusError")<{
  readonly message: string
  readonly cause?: unknown
}> {}

// Event handler type
export type EventHandler<E extends DomainEvent = DomainEvent> = (
  event: E
) => Effect.Effect<void, unknown>

// Event bus service
export class EventBus extends Context.Tag("EventBus")
  EventBus,
  {
    readonly publish: (event: DomainEvent) => Effect.Effect<void, EventBusError>
    readonly subscribe: <E extends DomainEvent>(
      eventType: E["_tag"],
      handler: EventHandler<E>
    ) => Effect.Effect<void, EventBusError>
    readonly subscribeAll: (
      handler: EventHandler
    ) => Effect.Effect<void, EventBusError>
  }
>() {}

export const makeEventBus = Effect.gen(function* () {
  const eventQueue = yield* Queue.unbounded<DomainEvent>()
  const handlers = yield* Ref.make<Map<string, Array<EventHandler>>>(new Map())
  const globalHandlers = yield* Ref.make<Array<EventHandler>>([])

  // Start event processor
  const processor = Stream.fromQueue(eventQueue).pipe(
    Stream.tap((event) =>
      Effect.gen(function* () {
        yield* Effect.log("Processing event", { eventType: event._tag })

        // Get specific handlers
        const handlerMap = yield* Ref.get(handlers)
        const specificHandlers = handlerMap.get(event._tag) || []

        // Get global handlers
        const allHandlers = yield* Ref.get(globalHandlers)

        // Execute all handlers
        yield* Effect.all(
          [...specificHandlers, ...allHandlers].map(handler =>
            handler(event).pipe(
              Effect.catchAll((error) =>
                Effect.log("Handler error", { error, event: event._tag })
              )
            )
          ),
          { concurrency: "unbounded", discard: true }
        )
      })
    ),
    Stream.runDrain,
    Effect.forkDaemon
  )

  yield* processor

  const publish = (event: DomainEvent): Effect.Effect<void, EventBusError> =>
    Queue.offer(eventQueue, event).pipe(
      Effect.asVoid,
      Effect.mapError((cause) => new EventBusError({
        message: "Failed to publish event",
        cause
      }))
    )

  const subscribe = <E extends DomainEvent>(
    eventType: E["_tag"],
    handler: EventHandler<E>
  ): Effect.Effect<void, EventBusError> =>
    Ref.update(handlers, (current) => {
      const eventHandlers = current.get(eventType) || []
      return new Map(current).set(eventType, [...eventHandlers, handler as EventHandler])
    }).pipe(
      Effect.asVoid,
      Effect.mapError((cause) => new EventBusError({
        message: "Failed to subscribe",
        cause
      }))
    )

  const subscribeAll = (handler: EventHandler): Effect.Effect<void, EventBusError> =>
    Ref.update(globalHandlers, (current) => [...current, handler]).pipe(
      Effect.asVoid,
      Effect.mapError((cause) => new EventBusError({
        message: "Failed to subscribe to all events",
        cause
      }))
    )

  return { publish, subscribe, subscribeAll } as const
})

export const EventBusLive = Layer.effect(EventBus, makeEventBus)
```

### Event Sourcing Pattern

```typescript
// Event store service
export class EventStore extends Context.Tag("EventStore")
  EventStore,
  {
    readonly append: (
      aggregateId: string,
      events: ReadonlyArray<DomainEvent>
    ) => Effect.Effect<void, EventStoreError>
    readonly getEvents: (
      aggregateId: string,
      fromVersion?: number
    ) => Effect.Effect<ReadonlyArray<DomainEvent>, EventStoreError>
    readonly getAllEvents: Effect.Effect<ReadonlyArray<DomainEvent>, EventStoreError>
  }
>() {}

export class EventStoreError extends Data.TaggedError("EventStoreError")<{
  readonly message: string
  readonly cause?: unknown
}> {}

// In-memory implementation (use database in production)
export const makeEventStore = Effect.gen(function* () {
  const events = yield* Ref.make<Map<string, Array<DomainEvent>>>(new Map())

  const append = (
    aggregateId: string,
    newEvents: ReadonlyArray<DomainEvent>
  ): Effect.Effect<void, EventStoreError> =>
    Effect.gen(function* () {
      yield* Ref.update(events, (current) => {
        const aggregateEvents = current.get(aggregateId) || []
        return new Map(current).set(aggregateId, [...aggregateEvents, ...newEvents])
      })

      yield* Effect.log("Events appended", {
        aggregateId,
        eventCount: newEvents.length
      })
    }).pipe(
      Effect.mapError((cause) => new EventStoreError({
        message: "Failed to append events",
        cause
      }))
    )

  const getEvents = (
    aggregateId: string,
    fromVersion: number = 0
  ): Effect.Effect<ReadonlyArray<DomainEvent>, EventStoreError> =>
    Effect.gen(function* () {
      const allEvents = yield* Ref.get(events)
      const aggregateEvents = allEvents.get(aggregateId) || []

      return aggregateEvents.filter(event => event.version >= fromVersion)
    }).pipe(
      Effect.mapError((cause) => new EventStoreError({
        message: "Failed to get events",
        cause
      }))
    )

  const getAllEvents = Effect.gen(function* () {
    const allEvents = yield* Ref.get(events)
    return Array.from(allEvents.values()).flat()
  }).pipe(
    Effect.mapError((cause) => new EventStoreError({
      message: "Failed to get all events",
      cause
    }))
  )

  return { append, getEvents, getAllEvents } as const
})

export const EventStoreLive = Layer.effect(EventStore, makeEventStore)

// Aggregate reconstruction from events
export const rebuildOrderAggregate = (
  orderId: string
): Effect.Effect<OrderState, EventStoreError> =>
  Effect.gen(function* () {
    const eventStore = yield* EventStore
    const events = yield* eventStore.getEvents(orderId)

    // Start with initial state
    let state: OrderState = { _tag: "Pending", orderId }

    // Apply each event
    for (const event of events) {
      state = yield* applyEvent(state, event)
    }

    return state
  })

const applyEvent = (
  state: OrderState,
  event: DomainEvent
): Effect.Effect<OrderState, never> =>
  Match.value(event).pipe(
    Match.tag("OrderCreated", () => Effect.succeed(state)),
    Match.tag("OrderConfirmed", (evt) =>
      Effect.succeed({
        _tag: "Confirmed" as const,
        orderId: evt.aggregateId,
        paymentId: evt.paymentId
      })
    ),
    Match.tag("OrderShipped", (evt) =>
      Effect.succeed({
        _tag: "Shipped" as const,
        orderId: evt.aggregateId,
        trackingId: evt.trackingNumber
      })
    ),
    Match.tag("OrderCancelled", (evt) =>
      Effect.succeed({
        _tag: "Cancelled" as const,
        orderId: evt.aggregateId,
        reason: evt.reason
      })
    ),
    Match.exhaustive
  )
```

### Event Projection for Read Models

```typescript
// Projection service for building read models
export class OrderProjection extends Context.Tag("OrderProjection")
  OrderProjection,
  {
    readonly project: (event: DomainEvent) => Effect.Effect<void, ProjectionError>
    readonly getOrderSummary: (orderId: string) => Effect.Effect<OrderSummary, ProjectionError>
    readonly getAllSummaries: Effect.Effect<ReadonlyArray<OrderSummary>, ProjectionError>
  }
>() {}

export class ProjectionError extends Data.TaggedError("ProjectionError")<{
  readonly message: string
  readonly cause?: unknown
}> {}

// Read model
export interface OrderSummary {
  readonly orderId: string
  readonly customerId: string
  readonly status: string
  readonly totalAmount: number
  readonly itemCount: number
  readonly createdAt: Date
  readonly lastUpdated: Date
}

export const makeOrderProjection = Effect.gen(function* () {
  const summaries = yield* Ref.make<Map<string, OrderSummary>>(new Map())

  const project = (event: DomainEvent): Effect.Effect<void, ProjectionError> =>
    Match.value(event).pipe(
      Match.tag("OrderCreated", (evt) =>
        Ref.update(summaries, (current) =>
          new Map(current).set(evt.aggregateId, {
            orderId: evt.aggregateId,
            customerId: evt.customerId,
            status: "created",
            totalAmount: evt.totalAmount,
            itemCount: evt.items.length,
            createdAt: evt.timestamp,
            lastUpdated: evt.timestamp
          })
        )
      ),
      Match.tag("OrderConfirmed", (evt) =>
        Ref.update(summaries, (current) => {
          const summary = current.get(evt.aggregateId)
          if (!summary) return current

          return new Map(current).set(evt.aggregateId, {
            ...summary,
            status: "confirmed",
            lastUpdated: evt.timestamp
          })
        })
      ),
      Match.tag("OrderShipped", (evt) =>
        Ref.update(summaries, (current) => {
          const summary = current.get(evt.aggregateId)
          if (!summary) return current

          return new Map(current).set(evt.aggregateId, {
            ...summary,
            status: "shipped",
            lastUpdated: evt.timestamp
          })
        })
      ),
      Match.tag("OrderCancelled", (evt) =>
        Ref.update(summaries, (current) => {
          const summary = current.get(evt.aggregateId)
          if (!summary) return current

          return new Map(current).set(evt.aggregateId, {
            ...summary,
            status: "cancelled",
            lastUpdated: evt.timestamp
          })
        })
      ),
      Match.exhaustive
    ).pipe(
      Effect.asVoid,
      Effect.mapError((cause) => new ProjectionError({
        message: "Failed to project event",
        cause
      }))
    )

  const getOrderSummary = (orderId: string): Effect.Effect<OrderSummary, ProjectionError> =>
    Effect.gen(function* () {
      const current = yield* Ref.get(summaries)
      const summary = current.get(orderId)

      if (!summary) {
        return yield* Effect.fail(new ProjectionError({
          message: `Order ${orderId} not found in projection`
        }))
      }

      return summary
    })

  const getAllSummaries = Effect.gen(function* () {
    const current = yield* Ref.get(summaries)
    return Array.from(current.values())
  })

  return { project, getOrderSummary, getAllSummaries } as const
})

export const OrderProjectionLive = Layer.effect(OrderProjection, makeOrderProjection)
```

## 16.3 CQRS Patterns

Command Query Responsibility Segregation separates read and write operations for scalability and optimization.

### Command Side Implementation

```typescript
import { Effect, Context, Layer, Data } from "effect"

// Command definitions
export class CreateOrderCommand extends Data.TaggedClass("CreateOrderCommand")<{
  readonly orderId: string
  readonly customerId: string
  readonly items: ReadonlyArray<{
    readonly productId: string
    readonly quantity: number
    readonly price: number
  }>
}> {}

export class ConfirmOrderCommand extends Data.TaggedClass("ConfirmOrderCommand")<{
  readonly orderId: string
  readonly paymentId: string
}> {}

export class ShipOrderCommand extends Data.TaggedClass("ShipOrderCommand")<{
  readonly orderId: string
  readonly trackingNumber: string
  readonly carrier: string
}> {}

export type OrderCommand = CreateOrderCommand | ConfirmOrderCommand | ShipOrderCommand

// Command errors
export class CommandValidationError extends Data.TaggedError("CommandValidationError")<{
  readonly message: string
  readonly field?: string
}> {}

export class CommandExecutionError extends Data.TaggedError("CommandExecutionError")<{
  readonly message: string
  readonly cause?: unknown
}> {}

// Command handler service
export class OrderCommandHandler extends Context.Tag("OrderCommandHandler")
  OrderCommandHandler,
  {
    readonly handle: (
      command: OrderCommand
    ) => Effect.Effect
      ReadonlyArray<DomainEvent>,
      CommandValidationError | CommandExecutionError
    >
  }
>() {}

export const makeOrderCommandHandler = Effect.gen(function* () {
  const eventStore = yield* EventStore
  const eventBus = yield* EventBus
  const inventoryService = yield* InventoryService

  const validateCreateOrder = (
    command: CreateOrderCommand
  ): Effect.Effect<void, CommandValidationError> =>
    Effect.gen(function* () {
      if (command.items.length === 0) {
        return yield* Effect.fail(new CommandValidationError({
          message: "Order must contain at least one item",
          field: "items"
        }))
      }

      const totalAmount = command.items.reduce((sum, item) =>
        sum + (item.price * item.quantity), 0
      )

      if (totalAmount <= 0) {
        return yield* Effect.fail(new CommandValidationError({
          message: "Order total must be greater than zero",
          field: "totalAmount"
        }))
      }
    })

  const handleCreateOrder = (
    command: CreateOrderCommand
  ): Effect.Effect
    ReadonlyArray<DomainEvent>,
    CommandValidationError | CommandExecutionError
  > =>
    Effect.gen(function* () {
      // Validate command
      yield* validateCreateOrder(command)

      // Check inventory availability
      yield* Effect.all(
        command.items.map(item =>
          inventoryService.checkAvailability(item.productId, item.quantity)
        ),
        { concurrency: 5 }
      ).pipe(
        Effect.mapError((cause) => new CommandExecutionError({
          message: "Inventory check failed",
          cause
        }))
      )

      // Create event
      const event = new OrderCreated({
        eventId: crypto.randomUUID(),
        eventType: "OrderCreated",
        timestamp: new Date(),
        aggregateId: command.orderId,
        version: 1,
        customerId: command.customerId,
        items: command.items,
        totalAmount: command.items.reduce((sum, item) =>
          sum + (item.price * item.quantity), 0
        )
      })

      // Persist and publish
      yield* eventStore.append(command.orderId, [event])
      yield* eventBus.publish(event)

      return [event]
    })

  const handleConfirmOrder = (
    command: ConfirmOrderCommand
  ): Effect.Effect
    ReadonlyArray<DomainEvent>,
    CommandValidationError | CommandExecutionError
  > =>
    Effect.gen(function* () {
      // Load current state
      const currentState = yield* rebuildOrderAggregate(command.orderId).pipe(
        Effect.mapError((cause) => new CommandExecutionError({
          message: "Failed to load order",
          cause
        }))
      )

      // Validate state allows confirmation
      if (currentState._tag !== "Pending" && currentState._tag !== "PaymentProcessing") {
        return yield* Effect.fail(new CommandValidationError({
          message: `Cannot confirm order in state: ${currentState._tag}`
        }))
      }

      // Create event
      const event = new OrderConfirmed({
        eventId: crypto.randomUUID(),
        eventType: "OrderConfirmed",
        timestamp: new Date(),
        aggregateId: command.orderId,
        version: 2,
        paymentId: command.paymentId,
        confirmedAt: new Date()
      })

      yield* eventStore.append(command.orderId, [event])
      yield* eventBus.publish(event)

      return [event]
    })

  const handle = (
    command: OrderCommand
  ): Effect.Effect
    ReadonlyArray<DomainEvent>,
    CommandValidationError | CommandExecutionError
  > =>
    Match.value(command).pipe(
      Match.tag("CreateOrderCommand", handleCreateOrder),
      Match.tag("ConfirmOrderCommand", handleConfirmOrder),
      Match.tag("ShipOrderCommand", (cmd) => Effect.succeed([]) as any),
      Match.exhaustive
    )

  return { handle } as const
})

export const OrderCommandHandlerLive = Layer.effect(
  OrderCommandHandler,
  makeOrderCommandHandler
)
```

```typescript
  readonly cause?: unknown

export interface OrderStatistics {

  readonly totalSpent: number

}

// Query handler service

  OrderQueryHandler,
  {
    readonly query: (
      query: OrderQuery
    ) => Effect.Effect<unknown, QueryExecutionError>
  }

export const makeOrderQueryHandler = Effect.gen(function* () {
  const projection = yield* OrderProjection

  const handleGetOrder = (
    query: GetOrderQuery
  ): Effect.Effect<OrderSummary, QueryExecutionError> =>
    projection.getOrderSummary(query.orderId).pipe(
      Effect.mapError((cause) => new QueryExecutionError({
        message: "Failed to get order",
        cause
      }))
    )

  const handleListOrders = (
    query: ListOrdersQuery
  ): Effect.Effect<ReadonlyArray<OrderSummary>, QueryExecutionError> =>
    Effect.gen(function* () {
      const allSummaries = yield* projection.getAllSummaries().pipe(
        Effect.mapError((cause) => new QueryExecutionError({
          message: "Failed to list orders",
          cause
        }))
      )

      // Filter by criteria
      let filtered = allSummaries

      if (query.customerId) {
        filtered = filtered.filter(s => s.customerId === query.customerId)
      }

      if (query.status) {
        filtered = filtered.filter(s => s.status === query.status)
      }

      // Apply pagination
      const offset = query.offset ?? 0
      const limit = query.limit ?? 50

      return filtered.slice(offset, offset + limit)
    })

  const handleGetStatistics = (
    query: GetOrderStatisticsQuery
  ): Effect.Effect<OrderStatistics, QueryExecutionError> =>
    Effect.gen(function* () {
      const allSummaries = yield* projection.getAllSummaries().pipe(
        Effect.mapError((cause) => new QueryExecutionError({
          message: "Failed to get statistics",
          cause
        }))
      )

      const customerOrders = allSummaries.filter(s =>
        s.customerId === query.customerId
      )

      const totalOrders = customerOrders.length
      const totalSpent = customerOrders.reduce((sum, order) =>
        sum + order.totalAmount, 0
      )
      const averageOrderValue = totalOrders > 0 ? totalSpent / totalOrders : 0

      const ordersByStatus = customerOrders.reduce((acc, order) => ({
        ...acc,
        [order.status]: (acc[order.status] || 0) + 1
      }), {} as Record<string, number>)

      return {
        totalOrders,
        totalSpent,
        averageOrderValue,
        ordersByStatus
      }
    })

  const query = (
    q: OrderQuery
  ): Effect.Effect<unknown, QueryExecutionError> =>
    Match.value(q).pipe(
      Match.tag("GetOrderQuery", handleGetOrder),
      Match.tag("ListOrdersQuery", handleListOrders),
      Match.tag("GetOrderStatisticsQuery", handleGetStatistics),
      Match.exhaustive
    )

  return { query } as const
})

export const OrderQueryHandlerLive = Layer.effect(
  OrderQueryHandler,
  makeOrderQueryHandler
)
```

## 16.4 Saga Patterns

Sagas coordinate distributed transactions across multiple services, ensuring eventual consistency through compensation.

### Orchestration-Based Saga

```typescript
import { Effect, Context, Layer, Data, Match } from "effect"

// Saga step definition
export interface SagaStep<I, O, E> {
  readonly name: string
  readonly execute: (input: I) => Effect.Effect<O, E>
  readonly compensate: (input: I, output: O) => Effect.Effect<void, E>
}

// Saga errors
export class SagaExecutionError extends Data.TaggedError("SagaExecutionError")<{
  readonly step: string
  readonly cause: unknown
}> {}

export class SagaCompensationError extends Data.TaggedError("SagaCompensationError")<{
  readonly step: string
  readonly cause: unknown
}> {}

// Saga orchestrator
export class OrderSaga extends Context.Tag("OrderSaga")
  OrderSaga,
  {
    readonly execute: (
      order: CreateOrderCommand
    ) => Effect.Effect
      { orderId: string; paymentId: string },
      SagaExecutionError | SagaCompensationError
    >
  }
>() {}

export const makeOrderSaga = Effect.gen(function* () {
  const inventoryService = yield* InventoryService
  const paymentService = yield* PaymentService
  const orderService = yield* OrderService
  const logger = yield* Logger

  // Step 1: Reserve inventory
  const reserveInventoryStep: SagaStep
    CreateOrderCommand,
    { reservationId: string },
    never
  > = {
    name: "reserve-inventory",
    execute: (input) =>
      Effect.gen(function* () {
        yield* logger.info("Reserving inventory", { orderId: input.orderId })

        const reservations = yield* Effect.all(
          input.items.map(item =>
            inventoryService.reserveStock({
              productId: item.productId,
              quantity: item.quantity,
              orderId: input.orderId
            })
          )
        )

        return { reservationId: reservations[0].id }
      }),
    compensate: (input, output) =>
      Effect.gen(function* () {
        yield* logger.info("Compensating inventory", { orderId: input.orderId })

        yield* Effect.all(
          input.items.map(item =>
            inventoryService.releaseStock({
              productId: item.productId,
              quantity: item.quantity,
              orderId: input.orderId
            })
          ),
          { discard: true }
        )
      })
  }

  // Step 2: Process payment
  const processPaymentStep: SagaStep
    CreateOrderCommand,
    { paymentId: string },
    never
  > = {
    name: "process-payment",
    execute: (input) =>
      Effect.gen(function* () {
        yield* logger.info("Processing payment", { orderId: input.orderId })

        const totalAmount = input.items.reduce(
          (sum, item) => sum + (item.price * item.quantity),
          0
        )

        const payment = yield* paymentService.charge({
          customerId: input.customerId,
          amount: totalAmount,
          orderId: input.orderId
        })

        return { paymentId: payment.id }
      }),
    compensate: (input, output) =>
      Effect.gen(function* () {
        yield* logger.info("Refunding payment", { orderId: input.orderId })

        yield* paymentService.refund({
          paymentId: output.paymentId,
          reason: "Order cancelled"
        })
      })
  }

  // Step 3: Create order
  const createOrderStep: SagaStep
    CreateOrderCommand & { paymentId: string },
    { orderId: string },
    never
  > = {
    name: "create-order",
    execute: (input) =>
      Effect.gen(function* () {
        yield* logger.info("Creating order", { orderId: input.orderId })

        yield* orderService.create({
          orderId: input.orderId,
          customerId: input.customerId,
          items: input.items,
          paymentId: input.paymentId
        })

        return { orderId: input.orderId }
      }),
    compensate: (input, output) =>
      Effect.gen(function* () {
        yield* logger.info("Cancelling order", { orderId: input.orderId })

        yield* orderService.cancel({
          orderId: output.orderId,
          reason: "Saga compensation"
        })
      })
  }

  // Execute saga with compensation
  const execute = (
    order: CreateOrderCommand
  ): Effect.Effect
    { orderId: string; paymentId: string },
    SagaExecutionError | SagaCompensationError
  > =>
    Effect.gen(function* () {
      const completedSteps: Array<{
        step: SagaStep<any, any, any>
        input: any
        output: any
      }> = []

      try {
        // Execute step 1
        const inventory = yield* reserveInventoryStep.execute(order).pipe(
          Effect.mapError((cause) => new SagaExecutionError({
            step: reserveInventoryStep.name,
            cause
          }))
        )
        completedSteps.push({
          step: reserveInventoryStep,
          input: order,
          output: inventory
        })

        // Execute step 2
        const payment = yield* processPaymentStep.execute(order).pipe(
          Effect.mapError((cause) => new SagaExecutionError({
            step: processPaymentStep.name,
            cause
          }))
        )
        completedSteps.push({
          step: processPaymentStep,
          input: order,
          output: payment
        })

        // Execute step 3
        const orderResult = yield* createOrderStep.execute({
          ...order,
          paymentId: payment.paymentId
        }).pipe(
          Effect.mapError((cause) => new SagaExecutionError({
            step: createOrderStep.name,
            cause
          }))
        )
        completedSteps.push({
          step: createOrderStep,
          input: { ...order, paymentId: payment.paymentId },
          output: orderResult
        })

        return {
          orderId: orderResult.orderId,
          paymentId: payment.paymentId
        }
      } catch (error) {
        // Compensate in reverse order
        yield* logger.error("Saga failed, starting compensation", { error })

        yield* Effect.all(
          completedSteps.reverse().map(({ step, input, output }) =>
            step.compensate(input, output).pipe(
              Effect.tapError((cause) =>
                logger.error("Compensation failed", {
                  step: step.name,
                  cause
                })
              ),
              Effect.mapError((cause) => new SagaCompensationError({
                step: step.name,
                cause
              }))
            )
          ),
          { concurrency: 1 } // Sequential compensation
        )

        throw error
      }
    })

  return { execute } as const
})

export const OrderSagaLive = Layer.effect(OrderSaga, makeOrderSaga)
```

```typescript
// Choreography using event-driven coordination
export const setupOrderChoreography = Effect.gen(function* () {

  const paymentService = yield* PaymentService

          inventoryService.reserveStock({

            quantity: item.quantity,

          eventId: crypto.randomUUID(),
          eventType: "InventoryReserved",
          timestamp: new Date(),
          aggregateId: event.aggregateId,

            reason: "Inventory reservation failed",
            cancelledAt: new Date()

        )
      )
    )
  )

  // Handler: Process payment when inventory reserved
  yield* eventBus.subscribe("InventoryReserved", (event) =>
    Effect.gen(function* () {
      const order = yield* orderService.getOrder(event.aggregateId)

      const payment = yield* paymentService.charge({
        customerId: order.customerId,

          timestamp: new Date(),
          aggregateId: event.aggregateId,
          version: event.version + 1,

      )

      Effect.catchAll((error) =>
        Effect.gen(function* () {
          // Release inventory on payment failure
          yield* inventoryService.releaseReservation(event.aggregateId)

          yield* eventBus.publish(
            new OrderCancelled({
              eventId: crypto.randomUUID(),
              eventType: "OrderCancelled",
              timestamp: new Date(),
              aggregateId: event.aggregateId,
              version: event.version + 1,
              reason: "Payment processing failed",
              cancelledAt: new Date()
            })
          )
        })
      )
    )
  )

  // Handler: Confirm order when payment processed
  yield* eventBus.subscribe("PaymentProcessed", (event) =>

          timestamp: new Date(),
```

### Saga State Persistence

```typescript
// Saga state tracking
export interface SagaState {
  readonly sagaId: string
  readonly status: "running" | "completed" | "compensating" | "failed"
  readonly currentStep: number
  readonly completedSteps: ReadonlyArray<{
    readonly stepName: string
    readonly output: unknown
  }>
  readonly startedAt: Date
  readonly completedAt?: Date
}

export class SagaStateStore extends Context.Tag("SagaStateStore")
  SagaStateStore,
  {
    readonly save: (state: SagaState) => Effect.Effect<void>
    readonly load: (sagaId: string) => Effect.Effect<SagaState>
    readonly updateStatus: (
      sagaId: string,
      status: SagaState["status"]
    ) => Effect.Effect<void>
  }
>() {}

// Persistent saga orchestrator
export const makePersistentSaga = Effect.gen(function* () {
  const stateStore = yield* SagaStateStore
  const logger = yield* Logger

  const executeWithPersistence = <I, O>(
    sagaId: string,
    steps: ReadonlyArray<SagaStep<I, any, any>>,
    input: I
  ): Effect.Effect<O, SagaExecutionError | SagaCompensationError> =>
    Effect.gen(function* () {
      // Initialize state
      yield* stateStore.save({
        sagaId,
        status: "running",
        currentStep: 0,
        completedSteps: [],
        startedAt: new Date()
      })

      const completedSteps: Array<any> = []

      try {
        // Execute each step
        for (let i = 0; i < steps.length; i++) {
          const step = steps[i]

          yield* logger.info(`Executing step ${i + 1}/${steps.length}`, {
            sagaId,
            step: step.name
          })

          const output = yield* step.execute(input).pipe(
            Effect.mapError((cause) => new SagaExecutionError({
              step: step.name,
              cause
            }))
          )

          completedSteps.push({ stepName: step.name, output })

          // Persist progress
          yield* stateStore.save({
            sagaId,
            status: "running",
            currentStep: i + 1,
            completedSteps,
            startedAt: new Date()
          })
        }

        // Mark as completed
        yield* stateStore.updateStatus(sagaId, "completed")

        return completedSteps[completedSteps.length - 1].output
      } catch (error) {
        yield* stateStore.updateStatus(sagaId, "compensating")

        // Compensate in reverse order
        for (let i = completedSteps.length - 1; i >= 0; i--) {
          const { stepName, output } = completedSteps[i]
          const step = steps.find(s => s.name === stepName)

          if (step) {
            yield* step.compensate(input, output).pipe(
              Effect.catchAll((cause) =>
                logger.error("Compensation failed", {
                  sagaId,
                  step: stepName,
                  cause
                })
              )
            )
          }
        }

        yield* stateStore.updateStatus(sagaId, "failed")
        throw error
      }
    })

  return { executeWithPersistence } as const
})
```

### Best Practices for Advanced Patterns

#### State Machines
1. **Explicit States**: Define all possible states as discriminated unions
2. **Type-Safe Transitions**: Use pattern matching for compile-time safety
3. **Guarded Transitions**: Implement business rule guards
4. **Side Effect Management**: Keep side effects explicit in actions
5. **Testing**: Test all valid and invalid transitions

#### Event-Driven Architecture
1. **Schema Validation**: Use Effect Schema for all events
2. **Event Versioning**: Plan for schema evolution from the start
3. **Idempotent Handlers**: Ensure handlers can safely retry
4. **Dead Letter Queues**: Handle poison messages gracefully
5. **Event Sourcing**: Store events as the source of truth
6. **Projections**: Build optimized read models from events

#### CQRS
1. **Separate Models**: Use different models for commands and queries
2. **Eventual Consistency**: Design for eventual consistency
3. **Command Validation**: Validate commands before execution
4. **Query Optimization**: Optimize read models for specific queries
5. **Event Publishing**: Publish events after successful command execution

#### Sagas
1. **Idempotent Operations**: Make all saga operations idempotent
2. **Compensation Logic**: Always implement compensation for each step
3. **State Persistence**: Persist saga state for recovery
4. **Timeout Handling**: Set appropriate timeouts for all operations
5. **Choose Pattern**: Use orchestration for complex flows, choreography for loose coupling
6. **Retry Strategy**: Implement exponential backoff with limits

These advanced patterns enable building sophisticated, resilient distributed systems with Effect-TS while maintaining type safety and testability throughout.
