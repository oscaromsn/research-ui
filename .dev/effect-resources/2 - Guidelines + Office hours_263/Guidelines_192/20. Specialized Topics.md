### 20.1 Library Development

#### Development Workflow

**MANDATORY Process for Effect Library Development:**

1. **Research Phase**
   - Understand existing codebase patterns
   - Review test files and usage examples
   - Identify related modules and dependencies
   - Use multiple approaches for complex problems

2. **Planning Phase**
   - Create detailed implementation plan
   - Identify validation checkpoints
   - Consider edge cases and error handling
   - Validate plan before implementation

3. **Implementation Phase**
   - Execute with frequent validation
   - **🚨 CRITICAL**: IMMEDIATELY run `pnpm lint --fix <typescript_file.ts>` after editing ANY TypeScript file
   - Run automated checks at each step
   - Stop and reassess if stuck

**🚨 MANDATORY FUNCTION DEVELOPMENT WORKFLOW:**

Follow this EXACT sequence when creating ANY new function:

1. **Create function** - Write the implementation in TypeScript file
2. **Lint TypeScript file** - Run `pnpm lint --fix <typescript_file.ts>`
3. **Check compilation** - Run `pnpm tsc`
4. **Lint again** - Run `pnpm lint --fix <typescript_file.ts>` again
5. **Ensure compilation** - Run `pnpm tsc` again to double-check
6. **Write test** - Create comprehensive test for the function
7. **Compile & lint test** - Run `pnpm tsc` then `pnpm lint --fix <test_file.ts>`

**NEVER SKIP ANY STEP** - This workflow is MANDATORY for every function.

#### Type-Level Programming

**Conditional Types:**

```typescript
type IsString<T> = T extends string ? true : false
```

#### Branded Types Deep Dive

**Nominal Typing with Brands:**

```typescript
// Create a branded type
type Brand<K extends string> = {
  readonly __brand: K
}

type Branded<T, K extends string> = T & Brand<K>

// Define branded types
type UserId = Branded<string, "UserId">
type Email = Branded<string, "Email">

// Runtime brand checking
const isUserId = (value: string): value is UserId => {
  // Validation logic
  return value.startsWith("user_")
}

// Safe constructor
const makeUserId = (value: string): UserId | null => {
  return isUserId(value) ? value : null
}
```

**Schema-Based Branding:**

```typescript
import { Schema } from "@effect/schema"

// Create branded schema
export const UserId = Schema.String.pipe(
  Schema.brand("UserId")
)
export type UserId = Schema.Schema.Type<typeof UserId>

// Decode with validation
const parseUserId = Schema.decodeSync(UserId)

// Usage
const userId = parseUserId("user-123")  // UserId

```

**Type Safety Benefits:**

```typescript
// ❌ Without branding - easy to mix up
function getUser(id: string) { }
function getOrder(id: string) { }

const orderId = "order-123"

// ✅ With branding - compile-time safety
function getUserBranded(id: UserId) { }
function getOrderBranded(id: OrderId) { }

const orderIdBranded: OrderId = makeOrderId("order-123")

```

**Runtime Behavior:**

```typescript

// But preserved at compile time
```

#### Monadic Composition

**Functor Laws:**

```typescript
// Identity: map(id) = id
const identityLaw = <A>(fa: Effect.Effect<A>) => {
  const left = fa.pipe(Effect.map(a => a))
  const right = fa
  // left ≡ right
}

// Composition: map(f . g) = map(f) . map(g)
const compositionLaw = <A, B, C>(
  fa: Effect.Effect<A>,
  f: (a: A) => B,
  g: (b: B) => C
) => {
  const left = fa.pipe(Effect.map(a => g(f(a))))
  const right = fa.pipe(Effect.map(f), Effect.map(g))
  // left ≡ right
}
```

**Monad Laws:**

```typescript
// Left identity: flatMap(succeed(a), f) = f(a)
const leftIdentity = <A, B>(
  a: A,
  f: (a: A) => Effect.Effect<B>
) => {
  const left = Effect.succeed(a).pipe(Effect.flatMap(f))
  const right = f(a)
  // left ≡ right
}

// Right identity: flatMap(m, succeed) = m
const rightIdentity = <A>(m: Effect.Effect<A>) => {
  const left = m.pipe(Effect.flatMap(Effect.succeed))
  const right = m
  // left ≡ right
}

// Associativity: flatMap(flatMap(m, f), g) = flatMap(m, a => flatMap(f(a), g))
const associativity = <A, B, C>(
  m: Effect.Effect<A>,
  f: (a: A) => Effect.Effect<B>,
  g: (b: B) => Effect.Effect<C>
) => {
  const left = m.pipe(Effect.flatMap(f), Effect.flatMap(g))
  const right = m.pipe(Effect.flatMap(a => f(a).pipe(Effect.flatMap(g))))
  // left ≡ right
}
```

**flatMap/map/ap:**

```typescript
// map: Transform the value
const mapped = Effect.succeed(5).pipe(
  Effect.map(x => x * 2)
)  // Effect<10>

// flatMap: Transform and flatten
const flatMapped = Effect.succeed(5).pipe(
  Effect.flatMap(x => Effect.succeed(x * 2))
)  // Effect<10>

// ap: Apply a function in context
const fn = Effect.succeed((x: number) => x * 2)
const value = Effect.succeed(5)
const applied = fn.pipe(
  Effect.ap(value)
)  // Effect<10>
```

**Do Notation (gen):**

```typescript
// Do notation provides imperative-style syntax
const program = Effect.gen(function* () {
  // Bind values
  const a = yield* Effect.succeed(5)
  const b = yield* Effect.succeed(10)

  // Use them imperatively
  const sum = a + b

  // Conditional logic
  if (sum > 10) {
    yield* Effect.log("Sum is large")
  }

  return sum
})

// Equivalent to:
const programManual = Effect.succeed(5).pipe(
  Effect.flatMap(a =>
    Effect.succeed(10).pipe(
      Effect.flatMap(b =>
        Effect.sync(() => a + b).pipe(
          Effect.flatMap(sum =>
            sum > 10
              ? Effect.log("Sum is large").pipe(Effect.as(sum))
              : Effect.succeed(sum)
          )
        )
      )
    )
  )
)
```

#### Effect System Theory

**Effect as IO Monad:**

```typescript
// Effect is a description of a computation
type Effect<A, E = never, R = never> = {
  // Private implementation
}

// Like IO monad in Haskell:
// - Pure (referentially transparent)
// - Lazy (not executed until run)
// - Composable (using flatMap/map)
// - Type-safe (errors and requirements tracked)

// Pure value
const pure: Effect.Effect<number> = Effect.succeed(42)

// Impure operation described purely
const impure: Effect.Effect<string> = Effect.sync(() => {
  console.log("Side effect!")  // Not executed yet
  return "result"
})

// Execution happens at the edge
Effect.runPromise(impure)  // NOW the side effect happens
```

**Free Monads Concept:**

```typescript
// Free monad separates structure from interpretation
// Effect uses similar concepts internally

// Define operations
type Operation<A> =
  | { _tag: "Read"; key: string; next: (value: string) => Operation<A> }
  | { _tag: "Write"; key: string; value: string; next: () => Operation<A> }
  | { _tag: "Return"; value: A }

// Interpret operations
const interpret = <A>(op: Operation<A>): A => {
  switch (op._tag) {
    case "Read":
      const value = localStorage.getItem(op.key) ?? ""
      return interpret(op.next(value))
    case "Write":
      localStorage.setItem(op.key, op.value)
      return interpret(op.next())
    case "Return":
      return op.value
  }
}

// Effect does this internally, but with:
// - More operations
// - Better type safety
// - Optimized execution
// - Concurrent support
```

**Tagless Final Style:**

```typescript
// Tagless final: Define capabilities as interfaces
interface Storage<F> {
  readonly read: (key: string) => F
  readonly write: (key: string, value: string) => F

// Effect implementation

}

// Promise implementation
const storagePromise: Storage<Promise<string>> = {
  read: (key) => Promise.resolve(localStorage.getItem(key) ?? ""),
  write: (key, value) => Promise.resolve(localStorage.setItem(key, value))
}

// Use polymorphically
const program = <F>(storage: Storage<F>) => {
  // Works with any F that's a Monad
}
```
