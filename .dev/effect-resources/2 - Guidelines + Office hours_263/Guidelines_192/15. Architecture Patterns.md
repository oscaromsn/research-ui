## 15. Architecture Patterns

### Core Principle

Structure Effect-TS applications using layered architectures that enforce clear separation of concerns, promote testability, and maintain explicit dependency flows. All architectural patterns must leverage Effect's dependency injection system via `Context.Tag` and `Layer` composition.

### 15.1 Layered Architecture

#### The 3-Layer Pattern (General Purpose)

The fundamental architecture for Effect applications separates concerns into three distinct layers with unidirectional dependencies:

```
┌─────────────────────────────────────┐
│  HANDLER LAYER                      │  ← HTTP/API Interface
│  (Presentation/Interface)           │
├─────────────────────────────────────┤
│  USECASE LAYER                      │  ← Business Logic
│  (Application/Domain Logic)         │
├─────────────────────────────────────┤
│  STORE LAYER                        │  ← Data Access
│  (Data Access/Infrastructure)       │
└─────────────────────────────────────┘
```

**Dependency Rule**: Upper layers depend on lower layers, never the reverse.

##### Store Layer (Data Access)

The Store Layer encapsulates all data access logic, database queries, and external data source interactions.

```typescript
import { Context, Effect, Layer } from "effect"
import { Data } from "effect"

// Define Store-specific errors
class RecordNotFound extends Data.TaggedError("RecordNotFound")<{
  readonly table: string
  readonly id: string
}> {}

class DatabaseError extends Data.TaggedError("DatabaseError")<{
  readonly message: string
  readonly operation: string
  readonly cause?: unknown
}> {}

// Store service interface
class TodoStore extends Context.Tag("@app/TodoStore")
  TodoStore,
  {
    readonly getAll: Effect.Effect<ReadonlyArray<Todo>, DatabaseError>
    readonly getById: (id: string) => Effect.Effect<Todo, RecordNotFound | DatabaseError>
    readonly create: (data: CreateTodoData) => Effect.Effect<Todo, DatabaseError>
    readonly update: (id: string, data: UpdateTodoData) => Effect.Effect
      Todo,
      RecordNotFound | DatabaseError
    >
    readonly delete: (id: string) => Effect.Effect<void, RecordNotFound | DatabaseError>
  }
>() {}

// Store implementation
const makeTodoStore = Effect.gen(function* () {
  const db = yield* Database

  const getAll = Effect.tryPromise({
    try: () => db.query("SELECT * FROM todos"),
    catch: (error) => new DatabaseError({ 
      message: "Failed to fetch todos", 
      operation: "getAll",
      cause: error 
    })
  }).pipe(
    Effect.map(rows => rows.map(parseTodo))
  )

  const getById = (id: string) =>
    Effect.tryPromise({
      try: () => db.query("SELECT * FROM todos WHERE id = ?", [id]),
      catch: (error) => new DatabaseError({ 
        message: "Failed to fetch todo",
        operation: "getById",
        cause: error 
      })
    }).pipe(
      Effect.flatMap(row =>
        row
          ? Effect.succeed(parseTodo(row))
          : Effect.fail(new RecordNotFound({ table: "todos", id }))
      )
    )

  const create = (data: CreateTodoData) =>
    Effect.tryPromise({
      try: () => db.insert("todos", data),
      catch: (error) => new DatabaseError({ 
        message: "Failed to create todo",
        operation: "create",
        cause: error 
      })
    }).pipe(
      Effect.flatMap(id => getById(id))
    )

  const update = (id: string, data: UpdateTodoData) =>
    Effect.gen(function* () {
      // Verify exists
      yield* getById(id)
      
      yield* Effect.tryPromise({
        try: () => db.update("todos", id, data),
        catch: (error) => new DatabaseError({ 
          message: "Failed to update todo",
          operation: "update",
          cause: error 
        })
      })
      
      return yield* getById(id)
    })

  const _delete = (id: string) =>
    Effect.gen(function* () {
      // Verify exists
      yield* getById(id)
      
      yield* Effect.tryPromise({
        try: () => db.delete("todos", id),
        catch: (error) => new DatabaseError({ 
          message: "Failed to delete todo",
          operation: "delete",
          cause: error 
        })
      })
    })

  return TodoStore.of({
    getAll,
    getById,
    create,
    update,
    delete: _delete
  })
})

export const TodoStoreLive = Layer.effect(TodoStore, makeTodoStore)
```

**Store Layer Responsibilities:**
- Execute database queries
- Map database errors to domain errors
- Perform CRUD operations
- Handle data serialization/deserialization
- Manage database transactions

**Store Layer Does NOT:**
- Contain business logic
- Validate business rules
- Coordinate multiple operations
- Handle HTTP concerns

##### Usecase Layer (Business Logic)

The Usecase Layer implements business logic, orchestrates operations across stores, and enforces business rules.

```typescript
// Define business-specific errors
class TodoAlreadyCompleted extends Data.TaggedError("TodoAlreadyCompleted")<{
  readonly id: string
}> {}

class InvalidTodoData extends Data.TaggedError("InvalidTodoData")<{
  readonly errors: ReadonlyArray<string>
}> {}

// Usecase service interface
class TodoUsecase extends Context.Tag("@app/TodoUsecase")
  TodoUsecase,
  {
    readonly createTodo: (data: CreateTodoData) => Effect.Effect
      Todo,
      InvalidTodoData | DatabaseError
    >
    readonly completeTodo: (id: string) => Effect.Effect
      Todo,
      RecordNotFound | TodoAlreadyCompleted | DatabaseError
    >
    readonly getTodosByUser: (userId: string) => Effect.Effect
      ReadonlyArray<Todo>,
      DatabaseError
    >
    readonly archiveOldTodos: (olderThan: Date) => Effect.Effect
      number,
      DatabaseError
    >
  }
>() {}

// Usecase implementation
const makeTodoUsecase = Effect.gen(function* () {
  const store = yield* TodoStore
  const logger = yield* Logger
  const notificationService = yield* NotificationService

  const createTodo = (data: CreateTodoData) =>
    Effect.gen(function* () {
      // Validate business rules
      const errors = validateTodoData(data)
      if (errors.length > 0) {
        return yield* Effect.fail(new InvalidTodoData({ errors }))
      }

      yield* logger.info("Creating todo", { data })

      // Delegate to store
      const todo = yield* store.create(data)

      // Send notification (business logic)
      yield* notificationService.sendTodoCreated(todo)

      yield* logger.info("Todo created", { id: todo.id })
      
      return todo
    })

  const completeTodo = (id: string) =>
    Effect.gen(function* () {
      const todo = yield* store.getById(id)

      // Business rule: can't complete already completed todo
      if (todo.completed) {
        return yield* Effect.fail(new TodoAlreadyCompleted({ id }))
      }

      const updated = yield* store.update(id, {
        completed: true,
        completedAt: new Date()
      })

      // Business logic: award points
      yield* notificationService.sendTodoCompleted(updated)

      return updated
    })

  const getTodosByUser = (userId: string) =>
    store.getAll().pipe(
      Effect.map(todos => todos.filter(todo => todo.userId === userId)),
      Effect.tap(todos => 
        logger.debug("Fetched user todos", { 
          userId, 
          count: todos.length 
        })
      )
    )

  const archiveOldTodos = (olderThan: Date) =>
    Effect.gen(function* () {
      const allTodos = yield* store.getAll()
      
      const oldTodos = allTodos.filter(
        todo => todo.createdAt < olderThan && todo.completed
      )

      // Archive in parallel with concurrency limit
      yield* Effect.all(
        oldTodos.map(todo => store.delete(todo.id)),
        { concurrency: 10 }
      )

      yield* logger.info("Archived old todos", { count: oldTodos.length })

      return oldTodos.length
    })

  return TodoUsecase.of({
    createTodo,
    completeTodo,
    getTodosByUser,
    archiveOldTodos
  })
})

export const TodoUsecaseLive = Layer.effect(TodoUsecase, makeTodoUsecase)
```

**Usecase Layer Responsibilities:**
- Implement business logic
- Enforce business rules
- Coordinate multiple store operations
- Handle domain events
- Manage transactions across boundaries

**Usecase Layer Does NOT:**
- Execute database queries directly
- Handle HTTP request/response
- Contain presentation logic
- Manage UI state

##### Handler Layer (HTTP/API)

The Handler Layer manages HTTP concerns, request validation, response formatting, and error-to-HTTP mapping.

```typescript
import { 
  HttpServerRequest, 
  HttpServerResponse, 
  HttpApiBuilder 
} from "@effect/platform"

// Map domain errors to HTTP responses
const mapErrorToHttp = (error: unknown) => {
  if (error instanceof RecordNotFound) {
    return HttpServerResponse.json(
      { error: "Resource not found", details: error },
      { status: 404 }
    )
  }
  
  if (error instanceof InvalidTodoData) {
    return HttpServerResponse.json(
      { error: "Validation failed", errors: error.errors },
      { status: 400 }
    )
  }
  
  if (error instanceof TodoAlreadyCompleted) {
    return HttpServerResponse.json(
      { error: "Todo already completed", todoId: error.id },
      { status: 409 }
    )
  }
  
  if (error instanceof DatabaseError) {
    return HttpServerResponse.json(
      { error: "Internal server error" },
      { status: 500 }
    )
  }

  return HttpServerResponse.json(
    { error: "Unknown error" },
    { status: 500 }
  )
}

// Handler implementation
const getTodoHandler = (req: HttpServerRequest.HttpServerRequest) =>
  Effect.gen(function* () {
    const usecase = yield* TodoUsecase
    const { id } = yield* getPathParams(req)

    const todo = yield* usecase.getTodoById(id)

    return HttpServerResponse.json(todo)
  }).pipe(
    Effect.catchAll(mapErrorToHttp)
  )

const createTodoHandler = (req: HttpServerRequest.HttpServerRequest) =>
  Effect.gen(function* () {
    const usecase = yield* TodoUsecase

    // Parse and validate request body
    const body = yield* HttpServerRequest.schemaBodyJson(CreateTodoSchema)(req).pipe(
      Effect.mapError(() => 
        HttpServerResponse.json(
          { error: "Invalid request body" },
          { status: 400 }
        )
      )
    )

    const todo = yield* usecase.createTodo(body)

    return HttpServerResponse.json(todo, { status: 201 })
  }).pipe(
    Effect.catchAll(mapErrorToHttp)
  )

// Co-located usecase pattern (for simple operations)
const deleteTodoHandler = (req: HttpServerRequest.HttpServerRequest) =>
  Effect.gen(function* () {
    const store = yield* TodoStore
    const logger = yield* Logger
    const { id } = yield* getPathParams(req)

    yield* logger.info("Deleting todo", { id })
    
    yield* store.delete(id).pipe(
      Effect.orDie // Die on DatabaseError (log and crash)
    )

    return HttpServerResponse.empty({ status: 204 })
  }).pipe(
    Effect.catchAll(mapErrorToHttp)
  )
```

**Handler Layer Responsibilities:**
- Parse HTTP requests
- Validate request schemas
- Map errors to HTTP status codes
- Format responses
- Handle authentication/authorization
- Add tracing/logging metadata

**Handler Layer Does NOT:**
- Implement business logic
- Execute database queries
- Contain domain rules
- Manage domain state

#### The 7-Layer Pattern (Frontend/Svelte-Specific)

For frontend applications, especially with Svelte 5, a more detailed 7-layer architecture provides better separation:

```
┌─────────────────────────────────────┐
│  1. Components (UI)                 │  ← Svelte components
├─────────────────────────────────────┤
│  2. Composables (Reusable Logic)    │  ← UI logic & state management
├─────────────────────────────────────┤
│  3. Store Layer (State)             │  ← Svelte runes + Effect
├─────────────────────────────────────┤
│  4. Service Layer (API)             │  ← Effect services
├─────────────────────────────────────┤
│  5. Schema Validation               │  ← @effect/schema
├─────────────────────────────────────┤
│  6. Error Handling                  │  ← Tagged errors
├─────────────────────────────────────┤
│  7. Testing                         │  ← Test layer
└─────────────────────────────────────┘
```

##### Layer 2: Composables (Reusable UI Logic)

```typescript
// composables/useTodoForm.ts
import { Effect } from "effect"

export const useTodoForm = () => {
  let formData = $state<CreateTodoInput>({
    title: "",
    description: ""
  })
  
  let submitting = $state(false)
  let errors = $state<Record<string, string>>({})

  const submit = () => {
    submitting = true
    errors = {}

    Effect.gen(function* () {
      // Validate with schema
      const validated = yield* Schema.decode(CreateTodoSchema)(formData)
      
      // Call service
      const service = yield* TodoService
      return yield* service.createTodo(validated)
    }).pipe(
      Effect.runPromise
    ).then(
      (todo) => {
        submitting = false
        formData = { title: "", description: "" } // Reset form
      },
      (error) => {
        submitting = false
        if (error instanceof ValidationError) {
          errors = { [error.field]: error.message }
        }
      }
    )
  }

  return {
    get formData() { return formData },
    get submitting() { return submitting },
    get errors() { return errors },
    submit
  }
}
```

##### Layer 3: Store Layer (Svelte State + Effect)

```typescript
// stores/todos.store.ts
import { Effect } from "effect"

export const createTodosStore = () => {
  let todos = $state<ReadonlyArray<Todo>>([])
  let loading = $state(false)
  let error = $state<TodoError | null>(null)

  const fetchAll = () => {
    loading = true
    error = null

    Effect.gen(function* () {
      const service = yield* TodoService
      return yield* service.getAllTodos()
    }).pipe(
      Effect.runPromise
    ).then(
      (result) => {
        todos = result
        loading = false
      },
      (err) => {
        error = err
        loading = false
      }
    )
  }

  const createTodo = (input: CreateTodoInput) => {
    loading = true
    error = null

    return Effect.gen(function* () {
      const service = yield* TodoService
      const newTodo = yield* service.createTodo(input)
      return newTodo
    }).pipe(
      Effect.runPromise
    ).then(
      (newTodo) => {
        todos = [...todos, newTodo]
        loading = false
        return newTodo
      },
      (err) => {
        error = err
        loading = false
        throw err
      }
    )
  }

  return {
    get todos() { return todos },
    get loading() { return loading },
    get error() { return error },
    fetchAll,
    createTodo
  }
}
```

### 15.2 Domain-Driven Design

#### Domain Entities

Define rich domain entities using `Schema.Class` to encapsulate behavior and enforce invariants.

```typescript
import { Schema } from "@effect/schema"

// Value Object with validation
class EmailAddress extends Schema.Class<EmailAddress>("EmailAddress")({
  value: Schema.String.pipe(
    Schema.pattern(/^[^\s@]+@[^\s@]+\.[^\s@]+$/),
    Schema.brand("EmailAddress")
  )
}) {
  get domain(): string {
    return this.value.split("@")[1]
  }

  isCompanyEmail(company: string): boolean {
    return this.domain === company
  }
}

// Entity with rich behavior
class User extends Schema.Class<User>("User")({
  id: UserId,
  email: EmailAddress,
  name: Schema.NonEmptyString,
  role: Schema.Literal("admin", "user", "guest"),
  createdAt: Schema.DateTimeUtc,
  lastLoginAt: Schema.optionalWith(Schema.DateTimeUtc, { 
    default: () => undefined 
  })
}) {
  isAdmin(): boolean {
    return this.role === "admin"
  }

  hasLoggedInRecently(withinDays: number): boolean {
    if (!this.lastLoginAt) return false
    
    const daysSinceLogin = 
      (Date.now() - this.lastLoginAt.getTime()) / (1000 * 60 * 60 * 24)
    
    return daysSinceLogin <= withinDays
  }

  canAccessResource(resource: Resource): boolean {
    if (this.isAdmin()) return true
    return resource.ownerId === this.id
  }
}
```

#### Aggregates

Group related entities into aggregates with a single root and enforce consistency boundaries.

```typescript
// Aggregate root
class Order extends Schema.Class<Order>("Order")({
  id: OrderId,
  customerId: CustomerId,
  items: Schema.Array(OrderItem),
  status: Schema.Literal("pending", "confirmed", "shipped", "delivered"),
  totalAmount: Schema.Number,
  createdAt: Schema.DateTimeUtc,
  updatedAt: Schema.DateTimeUtc
}) {
  // Aggregate invariants
  static make(
    customerId: CustomerId,
    items: ReadonlyArray<OrderItem>
  ): Effect.Effect<Order, InvalidOrderError> {
    return Effect.gen(function* () {
      if (items.length === 0) {
        return yield* Effect.fail(
          new InvalidOrderError({ reason: "Order must have at least one item" })
        )
      }

      const totalAmount = items.reduce((sum, item) => 
        sum + item.price * item.quantity, 
        0
      )

      if (totalAmount <= 0) {
        return yield* Effect.fail(
          new InvalidOrderError({ reason: "Order total must be positive" })
        )
      }

      return new Order({
        id: OrderId.generate(),
        customerId,
        items,
        status: "pending",
        totalAmount,
        createdAt: DateTime.unsafeNow(),
        updatedAt: DateTime.unsafeNow()
      })
    })
  }

  // Business operations
  confirm(): Effect.Effect<Order, OrderAlreadyConfirmedError> {
    if (this.status !== "pending") {
      return Effect.fail(
        new OrderAlreadyConfirmedError({ orderId: this.id })
      )
    }

    return Effect.succeed(
      new Order({ ...this, status: "confirmed", updatedAt: DateTime.unsafeNow() })
    )
  }

  ship(): Effect.Effect<Order, OrderNotConfirmedError> {
    if (this.status !== "confirmed") {
      return Effect.fail(
        new OrderNotConfirmedError({ orderId: this.id })
      )
    }

    return Effect.succeed(
      new Order({ ...this, status: "shipped", updatedAt: DateTime.unsafeNow() })
    )
  }
}
```

#### Domain Services

Encapsulate domain logic that doesn't naturally belong to a single entity.

```typescript
// Domain service for business logic spanning multiple aggregates
class OrderDomainService extends Context.Tag("@domain/OrderDomainService")
  OrderDomainService,
  {
    readonly calculateDiscount: (
      order: Order,
      customer: Customer
    ) => Effect.Effect<number>
    
    readonly validateOrderForCustomer: (
      order: Order,
      customer: Customer
    ) => Effect.Effect<void, OrderValidationError>
  }
>() {}

const makeOrderDomainService = Effect.gen(function* () {
  const calculateDiscount = (order: Order, customer: Customer) =>
    Effect.gen(function* () {
      let discount = 0

      // VIP customer discount
      if (customer.isVip()) {
        discount += order.totalAmount * 0.10
      }

      // First-time customer discount
      if (customer.orderCount === 0) {
        discount += order.totalAmount * 0.05
      }

      // Bulk order discount
      if (order.items.length >= 10) {
        discount += order.totalAmount * 0.15
      }

      return Math.min(discount, order.totalAmount * 0.30) // Max 30%
    })

  const validateOrderForCustomer = (order: Order, customer: Customer) =>
    Effect.gen(function* () {
      // Check customer credit limit
      if (order.totalAmount > customer.creditLimit) {
        return yield* Effect.fail(
          new OrderValidationError({ 
            reason: "Order exceeds customer credit limit" 
          })
        )
      }

      // Check customer address
      if (!customer.hasValidShippingAddress()) {
        return yield* Effect.fail(
          new OrderValidationError({ 
            reason: "Customer has no valid shipping address" 
          })
        )
      }
    })

  return OrderDomainService.of({
    calculateDiscount,
    validateOrderForCustomer
  })
})

export const OrderDomainServiceLive = Layer.effect(
  OrderDomainService,
  makeOrderDomainService
)
```

#### Domain Events

Model domain events to capture significant business occurrences.

```typescript
// Domain event types
class OrderCreated extends Schema.TaggedClass<OrderCreated>()(
  "OrderCreated",
  {
    orderId: OrderId,
    customerId: CustomerId,
    totalAmount: Schema.Number,
    occurredAt: Schema.DateTimeUtc
  }
) {}

class OrderConfirmed extends Schema.TaggedClass<OrderConfirmed>()(
  "OrderConfirmed",
  {
    orderId: OrderId,
    confirmedBy: UserId,
    occurredAt: Schema.DateTimeUtc
  }
) {}

class OrderShipped extends Schema.TaggedClass<OrderShipped>()(
  "OrderShipped",
  {
    orderId: OrderId,
    trackingNumber: Schema.String,
    carrier: Schema.String,
    occurredAt: Schema.DateTimeUtc
  }
) {}

// Domain event union
type OrderEvent = OrderCreated | OrderConfirmed | OrderShipped

// Event handler service
class OrderEventHandler extends Context.Tag("@domain/OrderEventHandler")
  OrderEventHandler,
  {
    readonly handle: (event: OrderEvent) => Effect.Effect<void, EventHandlingError>
  }
>() {}

const makeOrderEventHandler = Effect.gen(function* () {
  const logger = yield* Logger
  const notificationService = yield* NotificationService

  const handle = (event: OrderEvent) =>
    Effect.gen(function* () {
      switch (event._tag) {
        case "OrderCreated":
          yield* logger.info("Order created", { orderId: event.orderId })
          yield* notificationService.sendOrderCreated(event)
          break

        case "OrderConfirmed":
          yield* logger.info("Order confirmed", { orderId: event.orderId })
          yield* notificationService.sendOrderConfirmed(event)
          break

        case "OrderShipped":
          yield* logger.info("Order shipped", { orderId: event.orderId })
          yield* notificationService.sendOrderShipped(event)
          break
      }
    })

  return OrderEventHandler.of({ handle })
})

export const OrderEventHandlerLive = Layer.effect(
  OrderEventHandler,
  makeOrderEventHandler
)
```

### 15.3 Dependency Management

#### Dependency Injection with Context.Tag

Use Effect's `Context.Tag` and `Layer` system for explicit, type-safe dependency injection.

```typescript
// Define dependencies
class Database extends Context.Tag("@app/Database")
  Database,
  { readonly query: (sql: string) => Effect.Effect<unknown> }
>() {}

class Logger extends Context.Tag("@app/Logger")
  Logger,
  {
    readonly info: (message: string) => Effect.Effect<void>
    readonly error: (message: string) => Effect.Effect<void>
  }
>() {}

class Cache extends Context.Tag("@app/Cache")
  Cache,
  {
    readonly get: (key: string) => Effect.Effect<Option.Option<unknown>>
    readonly set: (key: string, value: unknown) => Effect.Effect<void>
  }
>() {}

// Service with explicit dependencies
class UserService extends Context.Tag("@app/UserService")
  UserService,
  {
    readonly getUser: (id: string) => Effect.Effect<User, GetUserError>
  }
>() {}

const makeUserService = Effect.gen(function* () {
  // Explicitly yield dependencies
  const db = yield* Database
  const logger = yield* Logger
  const cache = yield* Cache

  const getUser = (id: string) =>
    Effect.gen(function* () {
      // Try cache first
      const cached = yield* cache.get(`user:${id}`)
      if (Option.isSome(cached)) {
        yield* logger.info(`Cache hit for user ${id}`)
        return cached.value as User
      }

      // Fetch from database
      yield* logger.info(`Fetching user ${id} from database`)
      const user = yield* db.query(`SELECT * FROM users WHERE id = '${id}'`)

      // Cache the result
      yield* cache.set(`user:${id}`, user)

      return user as User
    })

  return UserService.of({ getUser })
})

// Layer explicitly lists dependencies
export const UserServiceLive = Layer.effect(UserService, makeUserService)
```

#### Dependency Inversion Principle

Depend on abstractions (interfaces) rather than concrete implementations.

```typescript
// ✅ GOOD: Abstract interface
class PaymentGateway extends Context.Tag("@app/PaymentGateway")
  PaymentGateway,
  {
    readonly processPayment: (
      amount: number,
      currency: string
    ) => Effect.Effect<PaymentResult, PaymentError>
  }
>() {}

// Multiple implementations
const StripeGatewayLive = Layer.succeed(PaymentGateway, {
  processPayment: (amount, currency) =>
    Effect.gen(function* () {
      // Stripe-specific implementation
      yield* Effect.log("Processing payment via Stripe")
      return { transactionId: "stripe-123", status: "success" }
    })
})

const PayPalGatewayLive = Layer.succeed(PaymentGateway, {
  processPayment: (amount, currency) =>
    Effect.gen(function* () {
      // PayPal-specific implementation
      yield* Effect.log("Processing payment via PayPal")
      return { transactionId: "paypal-456", status: "success" }
    })
})

// Service depends on abstraction, not implementation
class OrderService extends Context.Tag("@app/OrderService")
  OrderService,
  {
    readonly checkout: (order: Order) => Effect.Effect<void, CheckoutError>
  }
>() {}

const makeOrderService = Effect.gen(function* () {
  const paymentGateway = yield* PaymentGateway // Abstract dependency

  const checkout = (order: Order) =>
    Effect.gen(function* () {
      const result = yield* paymentGateway.processPayment(
        order.totalAmount,
        order.currency
      )

      yield* Effect.log("Payment processed", { 
        transactionId: result.transactionId 
      })
    })

  return OrderService.of({ checkout })
})

// Can swap implementations via layer composition
const ProductionLayer = Layer.provide(
  OrderServiceLive,
  StripeGatewayLive
)

const TestLayer = Layer.provide(
  OrderServiceLive,
  PayPalGatewayLive
)
```

#### Circular Dependencies

Detect and resolve circular dependencies through architectural refactoring.

```typescript
// ❌ BAD: Circular dependency
class ServiceA extends Context.Tag("ServiceA")
  ServiceA,
  { readonly methodA: () => Effect.Effect<void> }
>() {}

class ServiceB extends Context.Tag("ServiceB")
  ServiceB,
  { readonly methodB: () => Effect.Effect<void> }
>() {}

const makeServiceA = Effect.gen(function* () {
  const serviceB = yield* ServiceB // A depends on B
  
  const methodA = () =>
    Effect.gen(function* () {
      yield* serviceB.methodB()
    })

  return ServiceA.of({ methodA })
})

const makeServiceB = Effect.gen(function* () {
  const serviceA = yield* ServiceA // B depends on A - CIRCULAR!
  
  const methodB = () =>
    Effect.gen(function* () {
      yield* serviceA.methodA()
    })

  return ServiceB.of({ methodB })
})

// ✅ GOOD: Extract shared logic to break cycle
class SharedService extends Context.Tag("SharedService")
  SharedService,
  { readonly sharedMethod: () => Effect.Effect<void> }
>() {}

const makeSharedService = Effect.gen(function* () {
  const sharedMethod = () =>
    Effect.gen(function* () {
      yield* Effect.log("Shared logic")
    })

  return SharedService.of({ sharedMethod })
})

const makeServiceA = Effect.gen(function* () {
  const shared = yield* SharedService // A depends on Shared
  
  const methodA = () =>
    Effect.gen(function* () {
      yield* shared.sharedMethod()
    })

  return ServiceA.of({ methodA })
})

const makeServiceB = Effect.gen(function* () {
  const shared = yield* SharedService // B depends on Shared
  
  const methodB = () =>
    Effect.gen(function* () {
      yield* shared.sharedMethod()
    })

  return ServiceB.of({ methodB })
})

// Now we have: SharedService <ServiceA, SharedService <ServiceB (no cycle)
```

#### Module Independence through Event Bus

Use an event bus for loose coupling between modules.

```typescript
// shared/events/EventBus.ts
class EventBus extends Context.Tag("@app/EventBus")
  EventBus,
  {
    readonly publish: <E extends DomainEvent>(
      event: E
    ) => Effect.Effect<void>
    readonly subscribe: <E extends DomainEvent>(
      eventType: string,
      handler: (event: E) => Effect.Effect<void>
    ) => Effect.Effect<void>
  }
>() {}

// users/services/UserUsecase.ts
const makeUserUsecase = Effect.gen(function* () {
  const eventBus = yield* EventBus
  const userStore = yield* UserStore

  const createUser = (data: CreateUserData) =>
    Effect.gen(function* () {
      const user = yield* userStore.create(data)

      // Publish event instead of direct coupling
      yield* eventBus.publish(
        new UserCreated({
          userId: user.id,
          email: user.email,
          occurredAt: DateTime.unsafeNow()
        })
      )

      return user
    })

  return UserUsecase.of({ createUser })
})

// orders/services/OrderUsecase.ts - Subscribes to events
const makeOrderUsecase = Effect.gen(function* () {
  const eventBus = yield* EventBus
  const orderStore = yield* OrderStore

  // Subscribe to user events
  yield* eventBus.subscribe("UserCreated", (event: UserCreated) =>
    Effect.gen(function* () {
      yield* Effect.log("New user created, initializing order history", {
        userId: event.userId
      })

      // Initialize order history for new user
      yield* orderStore.initializeForUser(event.userId)
    })
  )

  return OrderUsecase.of({
    // ... other methods
  })
})
```

#### Module Testing Independence

Each module should be testable in isolation using mock layers.

### Best Practices Summary

#### Layered Architecture
- **Clear Boundaries**: Each layer has well-defined responsibilities
- **Unidirectional Dependencies**: Upper layers depend on lower layers only
- **No Layer Skipping**: Don't bypass layers (e.g., Handler → Store directly)
- **Error Transformation**: Transform errors at layer boundaries
- **Testability**: Each layer testable in isolation

#### Domain-Driven Design
- **Rich Entities**: Encapsulate behavior in domain entities
- **Aggregate Roots**: Enforce consistency boundaries
- **Domain Services**: Extract logic spanning multiple entities
- **Value Objects**: Use for domain concepts with no identity
- **Domain Events**: Model significant business occurrences

#### Dependency Management
- **Explicit Dependencies**: Always declare dependencies in service constructors
- **Abstraction**: Depend on interfaces, not implementations
- **Composition Root**: Compose all layers at application entry point
- **No Circular Dependencies**: Refactor architecture to eliminate cycles
- **Test Doubles**: Use Layer-based mocks for testing

#### Modular Architecture
- **Feature Modules**: Organize by business capability
- **Public API**: Expose minimal, stable interface
- **Loose Coupling**: Use event bus for cross-module communication
- **High Cohesion**: Keep related functionality together
- **Independent Testing**: Each module testable without others

### Anti-Patterns to Avoid

#### ❌ Layer Violations

```typescript
// ❌ BAD: Handler directly accessing Store (skipping Usecase)
const badHandler = Effect.gen(function* () {
  const store = yield* TodoStore // Violates layering
  return yield* store.getAll()
})

// ✅ GOOD: Handler uses Usecase
const goodHandler = Effect.gen(function* () {
  const usecase = yield* TodoUsecase
  return yield* usecase.getAllTodos()
})
```

#### ❌ Anemic Domain Model

```typescript
// ❌ BAD: Entity with no behavior
class User extends Schema.Class<User>("User")({
  id: UserId,
  name: Schema.String,
  role: Schema.String
}) {}

// All logic in service
const isUserAdmin = (user: User) => user.role === "admin"
const canUserAccess = (user: User, resource: Resource) => {
  return user.role === "admin" || resource.ownerId === user.id
}

// ✅ GOOD: Rich domain model
class User extends Schema.Class<User>("User")({
  id: UserId,
  name: Schema.String,
  role: Schema.String
}) {
  isAdmin(): boolean {
    return this.role === "admin"
  }

  canAccess(resource: Resource): boolean {
    return this.isAdmin() || resource.ownerId === this.id
  }
}
```

#### ❌ God Services

```typescript
// ❌ BAD: Monolithic service doing everything
class ApplicationService {
  readonly createUser: () => Effect.Effect<User>
  readonly sendEmail: () => Effect.Effect<void>
  readonly processPayment: () => Effect.Effect<PaymentResult>
  readonly generateReport: () => Effect.Effect<Report>
  readonly updateInventory: () => Effect.Effect<void>
  // ... 50 more methods
}

// ✅ GOOD: Focused services
class UserService {
  readonly createUser: () => Effect.Effect<User>
  readonly updateUser: () => Effect.Effect<User>
}

class EmailService {
  readonly send: () => Effect.Effect<void>
}

class PaymentService {
  readonly process: () => Effect.Effect<PaymentResult>
}
```

#### ❌ Tight Coupling Between Modules

```typescript
// ❌ BAD: Direct dependency on internal implementation
import { UserRepository } from "@/modules/users/infrastructure/UserRepository"

class OrderService {
  // Directly depends on infrastructure detail
  const repo = yield* UserRepository
}

// ✅ GOOD: Depend on public interface
import { UserService } from "@/modules/users"

class OrderService {
  // Depends on public API contract
  const userService = yield* UserService
}
```
