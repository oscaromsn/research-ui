## 10. Configuration Management

### Core Principle

All application configuration MUST be managed through Effect's `Config` module to ensure type safety, validation, and proper handling of sensitive values. Never access `process.env` directly outside of configuration definition.

### 10.1 Config Module

#### Basic Config Types

Effect provides strongly-typed configuration primitives for all common data types:

```typescript
import { Config, Effect } from "effect"

// String configuration
const apiKey = Config.string("API_KEY")

// Number configuration
const port = Config.number("PORT")

// Boolean configuration
const debugMode = Config.boolean("DEBUG_MODE")

// Secret configuration (redacted in logs)
const databasePassword = Config.secret("DB_PASSWORD")

// Duration configuration
const requestTimeout = Config.duration("REQUEST_TIMEOUT")

// Use in program
const program = Effect.gen(function* () {
  const key = yield* apiKey
  const portNum = yield* port
  const debug = yield* debugMode
  const password = yield* databasePassword
  const timeout = yield* requestTimeout
  
  yield* Effect.log("Configuration loaded", {
    port: portNum,
    debug,
    // Password is automatically redacted
    passwordSet: password.value.length > 0
  })
})
```

Combine multiple config values into structured objects:

```typescript
// Simple composition
const ServerConfig = Config.all({
```

#### Default Values

Provide fallback values for optional configuration:

```typescript
// Single default value
const port = Config.number("PORT").pipe(
  Config.withDefault(3000)

// Multiple defaults
const ServerConfig = Config.all({
  host: Config.string("SERVER_HOST").pipe(
    Config.withDefault("localhost")

  port: Config.number("SERVER_PORT").pipe(
    Config.withDefault(3000)

const getDefaultPort = (env: string): number => {
  switch (env) {
    case "production": return 80
    case "staging": return 8080
    default: return 3000
  }
}

const port = Effect.gen(function* () {
  const env = yield* Config.string("NODE_ENV").pipe(
    Config.withDefault("development")

// Required vs optional configuration

  apiKey: Config.string("API_KEY"), // Required - fails if missing

const OptionalConfig = Config.all({
```

### 10.2 Schema-Based Configuration

#### Schema Validation

Combine Config with Schema for runtime validation and type safety:

```typescript
import * as S from "@effect/schema/Schema"
import { Config, Effect } from "effect"

// Define config schema
const ServerConfigSchema = S.Struct({
  host: S.String,
  port: S.Number.pipe(
    S.int(),
    S.greaterThan(0),
    S.lessThanOrEqualTo(65535)
  ),
  protocol: S.Literal("http", "https"),
  corsOrigins: S.Array(S.String),
  maxConnections: S.Number.pipe(
    S.int(),
    S.greaterThan(0)
  )
})

type ServerConfig = S.Schema.Type<typeof ServerConfigSchema>

// Load and validate configuration
const loadServerConfig = Effect.gen(function* () {
  const rawConfig = yield* Config.all({
    host: Config.string("SERVER_HOST"),
    port: Config.number("SERVER_PORT"),
    protocol: Config.string("SERVER_PROTOCOL"),
    corsOrigins: Config.array(Config.string("CORS_ORIGINS")),
    maxConnections: Config.number("MAX_CONNECTIONS")
  })
  
  // Validate against schema
  const validatedConfig = yield* S.decodeUnknown(ServerConfigSchema)(rawConfig)
  
  return validatedConfig
})

// Alternative: Direct schema-based config
const ServerConfig = Config.all({
  host: Config.string("SERVER_HOST"),
  port: Config.number("SERVER_PORT").pipe(
    Config.validate({
      message: "Port must be between 1 and 65535",
      validation: (port) => port > 0 && port <= 65535
    })
  ),
  protocol: Config.string("SERVER_PROTOCOL").pipe(
    Config.validate({
      message: "Protocol must be http or https",
      validation: (p) => p === "http" || p === "https"
    })
  )
})
```

#### DatabaseConfig Pattern

Standard pattern for database configuration with connection pooling:

```typescript
// Define database configuration schema
const DatabaseConfigSchema = S.Struct({
  host: S.String,
  port: S.Number.pipe(S.int(), S.greaterThan(0)),
  database: S.String,
  username: S.String,
  password: S.String, // Use String here for schema validation
  maxConnections: S.Number.pipe(
    S.int(),
    S.greaterThan(0),
    S.lessThanOrEqualTo(100)
  ),
  minConnections: S.Number.pipe(
    S.int(),
    S.greaterThanOrEqualTo(0)
  ),
  idleTimeout: S.Number.pipe(S.int(), S.greaterThan(0)),
  connectionTimeout: S.Number.pipe(S.int(), S.greaterThan(0)),
  ssl: S.Boolean
})

type DatabaseConfig = S.Schema.Type<typeof DatabaseConfigSchema>

// Load database configuration with secrets
const loadDatabaseConfig = Effect.gen(function* () {
  const config = yield* Config.all({
    host: Config.string("DB_HOST"),
    port: Config.number("DB_PORT").pipe(
      Config.withDefault(5432)
    ),
    database: Config.string("DB_NAME"),
    username: Config.string("DB_USER"),
    password: Config.secret("DB_PASSWORD"), // Config.secret for sensitive data
    maxConnections: Config.number("DB_MAX_CONNECTIONS").pipe(
      Config.withDefault(20)
    ),
    minConnections: Config.number("DB_MIN_CONNECTIONS").pipe(
      Config.withDefault(2)
    ),
    idleTimeout: Config.number("DB_IDLE_TIMEOUT").pipe(
      Config.withDefault(30000)
    ),
    connectionTimeout: Config.number("DB_CONNECTION_TIMEOUT").pipe(
      Config.withDefault(5000)
    ),
    ssl: Config.boolean("DB_SSL").pipe(
      Config.withDefault(true)
    )
  })
  
  // Validate schema (convert secret to string for validation)
  const validatedConfig = yield* S.decodeUnknown(DatabaseConfigSchema)({
    ...config,
    password: config.password.value
  })
  
  // Return with secret preserved
  return {
    ...validatedConfig,
    password: config.password
  }
})

// Use in database service
const makeDatabaseService = Effect.gen(function* () {
  const config = yield* loadDatabaseConfig
  const logger = yield* Logger
  
  yield* logger.info("Initializing database connection", {
    host: config.host,
    port: config.port,
    database: config.database,
    username: config.username,
    maxConnections: config.maxConnections,
    ssl: config.ssl
    // Never log password
  })
  
  const pool = yield* Effect.acquireRelease(
    Effect.sync(() => createPool({
      host: config.host,
      port: config.port,
      database: config.database,
      user: config.username,
      password: config.password.value, // Extract secret only when needed
      max: config.maxConnections,
      min: config.minConnections,
      idleTimeoutMillis: config.idleTimeout,
      connectionTimeoutMillis: config.connectionTimeout,
      ssl: config.ssl
    })),
    (pool) => Effect.sync(() => pool.end())
  )
  
  return {
    query: <A>(sql: string, params?: unknown[]) =>
      Effect.tryPromise({
        try: () => pool.query(sql, params) as Promise<A>,
        catch: (error) => new DatabaseError({ cause: error })
      })
  } as const
})
```

#### ApiConfig Pattern

Standard pattern for API client configuration:

```typescript
// Define API configuration schema
const ApiConfigSchema = S.Struct({
  baseUrl: S.String.pipe(
    S.pattern(/^https?:\/\/.+/)
  ),
  apiKey: S.String,
  timeout: S.Number.pipe(
    S.int(),
    S.greaterThan(0)
  ),
  retryAttempts: S.Number.pipe(
    S.int(),
    S.greaterThanOrEqualTo(0),
    S.lessThanOrEqualTo(10)
  ),
  retryDelay: S.Number.pipe(
    S.int(),
    S.greaterThan(0)
  ),
  rateLimitPerSecond: S.Number.pipe(
    S.int(),
    S.greaterThan(0)
  )
})

type ApiConfig = S.Schema.Type<typeof ApiConfigSchema>

// Load API configuration
const loadApiConfig = Effect.gen(function* () {
  const config = yield* Config.all({
    baseUrl: Config.string("API_BASE_URL"),
    apiKey: Config.secret("API_KEY"),
    timeout: Config.number("API_TIMEOUT").pipe(
      Config.withDefault(30000)
    ),
    retryAttempts: Config.number("API_RETRY_ATTEMPTS").pipe(
      Config.withDefault(3)
    ),
    retryDelay: Config.number("API_RETRY_DELAY").pipe(
      Config.withDefault(1000)
    ),
    rateLimitPerSecond: Config.number("API_RATE_LIMIT").pipe(
      Config.withDefault(10)
    )
  })
  
  // Validate schema
  const validatedConfig = yield* S.decodeUnknown(ApiConfigSchema)({
    ...config,
    apiKey: config.apiKey.value
  })
  
  return {
    ...validatedConfig,
    apiKey: config.apiKey
  }
})

// Use in API client service
const makeApiClient = Effect.gen(function* () {
  const config = yield* loadApiConfig
  const logger = yield* Logger
  
  yield* logger.info("Initializing API client", {
    baseUrl: config.baseUrl,
    timeout: config.timeout,
    retryAttempts: config.retryAttempts,
    rateLimitPerSecond: config.rateLimitPerSecond
    // Never log apiKey
  })
  
  return {
    request: <A>(endpoint: string, options?: RequestOptions) =>
      Effect.gen(function* () {
        const response = yield* Effect.tryPromise({
          try: () => fetch(`${config.baseUrl}${endpoint}`, {
            ...options,
            headers: {
              ...options?.headers,
              'Authorization': `Bearer ${config.apiKey.value}`
            },
            signal: AbortSignal.timeout(config.timeout)
          }),
          catch: (error) => new ApiError({ cause: error })
        }).pipe(
          Effect.retry({
            times: config.retryAttempts,
            schedule: Schedule.exponential(`${config.retryDelay} millis`)
          })
        )
        
        return response
      })
  } as const
})
```

### 10.3 Sensitive Configuration

#### Config.secret

**CRITICAL**: Always use `Config.secret` for sensitive values to prevent accidental logging:

```typescript
import { Config, Effect, Logger } from "effect"

// ✅ CORRECT: Use Config.secret for sensitive data
const AppConfig = Config.all({
  apiKey: Config.secret("API_KEY"),
  databasePassword: Config.secret("DB_PASSWORD"),
  jwtSecret: Config.secret("JWT_SECRET"),
  encryptionKey: Config.secret("ENCRYPTION_KEY")
})

// Use secrets safely
const program = Effect.gen(function* () {
  const config = yield* AppConfig
  const logger = yield* Logger
  
  // ✅ CORRECT: Secrets are automatically redacted in logs
  yield* logger.info("Configuration loaded", { config })
  // Logs: { apiKey: "<redacted>", databasePassword: "<redacted>", ... }
  
  // ✅ CORRECT: Only extract secret value when actually needed
  const actualApiKey = config.apiKey.value
  
  // Use the secret value in operations
  yield* makeApiRequest(actualApiKey)
})

// ❌ WRONG: Never log secrets directly
const badProgram = Effect.gen(function* () {
  const config = yield* AppConfig
  const logger = yield* Logger
  
  // ❌ This exposes the secret!
  yield* logger.info("API Key", { key: config.apiKey.value })
})

// ❌ WRONG: Don't use plain Config.string for secrets
const BadConfig = Config.all({
  apiKey: Config.string("API_KEY"), // ❌ Will be logged!
  password: Config.string("PASSWORD") // ❌ Will be logged!
})
```

#### Safe Secret Handling

```typescript
// Pattern for safely using secrets
const useSecret = <A, E, R>(
  secret: ConfigSecret.ConfigSecret,
  use: (value: string) => Effect.Effect<A, E, R>
): Effect.Effect<A, E, R> =>
  Effect.sync(() => secret.value).pipe(
    Effect.flatMap(use)
  )

// Example: Database connection with secret
const connectDatabase = Effect.gen(function* () {
  const config = yield* Config.all({
    host: Config.string("DB_HOST"),
    username: Config.string("DB_USER"),
    password: Config.secret("DB_PASSWORD")
  })
  
  // Extract secret only in the connection function
  return yield* useSecret(config.password, (password) =>
    Effect.tryPromise({
      try: () => createConnection({
        host: config.host,
        user: config.username,
        password // Used only here, never logged
      }),
      catch: (error) => new ConnectionError({ cause: error })
    })
  )
})
```

```typescript
  Effect.gen(function* () {
    const value = secret.value

          message: "API key must be at least 32 characters"

      return yield* Effect.fail(

          message: "API key must be alphanumeric"

  })

      age: credentialAge,
      recommendation: "Rotate credentials soon"
    })
  }
})

// 4. Use least privilege principle
const restrictedConfig = Effect.gen(function* () {
  const role = yield* Config.string("USER_ROLE")

    })

      // No access to admin or database credentials
    })
  }
```

### 10.4 Configuration Providers

#### Custom ConfigProvider

Create custom providers for different configuration sources:

```typescript
import { ConfigProvider, Effect, Layer } from "effect"
import * as fs from "fs/promises"

// JSON file provider
const makeJsonConfigProvider = (filePath: string) =>
  Effect.gen(function* () {
    const content = yield* Effect.tryPromise({
      try: () => fs.readFile(filePath, "utf-8"),
      catch: (error) => new ConfigError({
        message: `Failed to read config file: ${filePath}`,
        cause: error
      })
    })
    
    const json = yield* Effect.try({
      try: () => JSON.parse(content),
      catch: (error) => new ConfigError({
        message: `Failed to parse config JSON`,
        cause: error
      })
    })
    
    // Flatten nested JSON into key-value pairs
    const flattenObject = (
      obj: Record<string, unknown>,
      prefix = ""
    ): Map<string, string> => {
      const result = new Map<string, string>()
      
      for (const [key, value] of Object.entries(obj)) {
        const fullKey = prefix ? `${prefix}_${key}` : key
        
        if (typeof value === "object" && value !== null && !Array.isArray(value)) {
          const nested = flattenObject(value as Record<string, unknown>, fullKey)
          for (const [k, v] of nested) {
            result.set(k, v)
          }
        } else {
          result.set(fullKey, String(value))
        }
      }
      
      return result
    }
    
    return ConfigProvider.fromMap(flattenObject(json))
  })

// Environment-specific provider
const makeEnvironmentConfigProvider = Effect.gen(function* () {
  const env = process.env.NODE_ENV || "development"
  
  switch (env) {
    case "production":
      return yield* makeJsonConfigProvider("./config/production.json")
    case "staging":
      return yield* makeJsonConfigProvider("./config/staging.json")
    case "test":
      return yield* makeJsonConfigProvider("./config/test.json")
    default:
      return yield* makeJsonConfigProvider("./config/development.json")
  }
})

// Layered provider (fallback chain)
const makeLayeredConfigProvider = Effect.gen(function* () {
  const envProvider = ConfigProvider.fromEnv()
  const fileProvider = yield* makeJsonConfigProvider("./config/defaults.json")
  
  // Try environment first, then file, then defaults
  return ConfigProvider.orElse(envProvider, fileProvider)
})

// Use custom provider
const program = Effect.gen(function* () {
  const provider = yield* makeEnvironmentConfigProvider
  
  const config = yield* Config.all({
    host: Config.string("SERVER_HOST"),
    port: Config.number("SERVER_PORT")
  }).pipe(
    Effect.provide(Layer.setConfigProvider(provider))
  )
  
  yield* Effect.log("Loaded configuration", { config })
})
```

```typescript
// Configuration verification helper
const verifyConfiguration = (config: ApplicationConfig) =>
  Effect.gen(function* () {
    const logger = yield* Logger
    
    // Verify required configuration is present
    const verifications = [
      {
        name: "Server host",
        value: config.server.host,
        valid: config.server.host.length > 0
      },
      {
        name: "Server port",
        value: config.server.port,

        })
      )
    }
    
    yield* logger.info("Configuration verification passed")
  })
```

```typescript
const ApplicationConfig = Config.all({
```

#### Feature Toggles

Implement feature flags for gradual rollouts and A/B testing:

```typescript
// Simple boolean feature flags
const FeatureFlags = Config.all({
  newUserInterface: Config.boolean("FEATURE_NEW_UI").pipe(
    Config.withDefault(false)
  ),
  advancedAnalytics: Config.boolean("FEATURE_ADVANCED_ANALYTICS").pipe(
    Config.withDefault(false)
  ),
  experimentalApi: Config.boolean("FEATURE_EXPERIMENTAL_API").pipe(
    Config.withDefault(false)
  )
})

// Percentage-based rollout
const PercentageFeatures = Config.all({
  newCheckoutFlow: Config.number("FEATURE_NEW_CHECKOUT_PERCENT").pipe(
    Config.withDefault(0),
    Config.validate({
      message: "Percentage must be between 0 and 100",
      validation: (n) => n >= 0 && n <= 100
    })
  )
})

// User-based features
const isFeatureEnabled = (
  userId: string,
  featurePercent: number
): boolean => {
  // Deterministic hash-based rollout
  const hash = hashString(userId)
  return (hash % 100) < featurePercent
}

// Environment-based features
const getEnvironmentFeatures = Effect.gen(function* () {
  const env = yield* Config.string("NODE_ENV").pipe(
    Config.withDefault("development")
  )
  
  return {
    enableDebugEndpoints: env === "development",
    enableStrictValidation: env === "production",
    enablePerformanceMonitoring: env === "production" || env === "staging"
  }
})

// Feature flag service
class FeatureFlagService extends Effect.Service<FeatureFlagService>()(
  "FeatureFlagService",
  {
    effect: Effect.gen(function* () {
      const flags = yield* FeatureFlags
      const percentages = yield* PercentageFeatures
      const envFeatures = yield* getEnvironmentFeatures
      
      const isEnabled = (feature: string, userId?: string): boolean => {
        // Check boolean flags
        if (feature in flags) {
          return flags[feature as keyof typeof flags]
        }
        
        // Check percentage-based rollout
        if (feature in percentages && userId) {
          const percent = percentages[feature as keyof typeof percentages]
          return isFeatureEnabled(userId, percent)
        }
        
        // Check environment-based features
        if (feature in envFeatures) {
          return envFeatures[feature as keyof typeof envFeatures]
        }
        
        return false
      }
      
      const getAll = () => ({
        ...flags,
        ...envFeatures
      })
      
      return { isEnabled, getAll } as const
    })
  }
) {}

// Usage in application
const handleRequest = (userId: string) =>
  Effect.gen(function* () {
    const features = yield* FeatureFlagService
    
    if (features.isEnabled("newUserInterface")) {
      return yield* renderNewUI(userId)
    }
    
    if (features.isEnabled("newCheckoutFlow", userId)) {
      return yield* newCheckoutProcess(userId)
    }
    
    return yield* defaultFlow(userId)
  })
```

### Best Practices Summary

1. **Always Use Config Module**: Never access `process.env` directly outside config definitions
2. **Type Safety**: Use schema validation for complex configuration
3. **Secret Handling**: Always use `Config.secret` for sensitive values
4. **Default Values**: Provide sensible defaults for optional configuration
5. **Environment Variables**: Use SCREAMING_SNAKE_CASE naming convention
6. **Fail Fast**: Validate configuration at startup and fail immediately on errors
7. **Never Log Secrets**: Be vigilant about not exposing sensitive configuration
8. **Documentation**: Maintain comprehensive documentation of all configuration options
9. **Nested Structure**: Organize complex configuration hierarchically
10. **Feature Flags**: Use feature toggles for gradual rollouts and experimentation
11. **Environment-Specific**: Load appropriate configuration based on environment
12. **Validation Messages**: Provide clear, actionable error messages for invalid configuration
