## 17. Platform Integration

### Overview

Platform integration in Effect-TS provides standardized patterns for connecting your application to databases, message queues, external services, and frontend frameworks. These integrations maintain Effect's core principles of type safety, composability, and explicit error handling while interfacing with the broader ecosystem.

**Core Integration Principles:**
- **Type Safety First**: All platform integrations must maintain end-to-end type safety
- **Service Encapsulation**: Wrap external platforms in Effect services for testability
- **Schema Validation**: Validate all data crossing platform boundaries with Schema
- **Resource Management**: Use proper lifecycle management for connections and resources
- **Error Handling**: Convert platform-specific errors to typed Effect errors

### 17.1 Database Integration

#### @effect/sql: Core Database Patterns

**Basic Query Execution:**

```typescript
import { Effect } from "effect"
import { SqlClient } from "@effect/sql"
import { PgClient } from "@effect/sql-pg"

// Define database errors
class DatabaseError extends Data.TaggedError("DatabaseError")<{
  cause: unknown
  operation: string
}> {}

// Execute queries with proper error handling
const fetchUser = (userId: string) =>
  Effect.gen(function* () {
    const sql = yield* SqlClient.SqlClient
    
    const result = yield* sql`
      SELECT id, name, email, created_at
      FROM users
      WHERE id = ${userId}
    `.pipe(
      Effect.mapError((cause) => 
        new DatabaseError({ cause, operation: "fetchUser" })
      )
    )
    
    return result.rows[0]
  })

// Prepared statements for better performance
const getUserById = (id: string) =>
  Effect.gen(function* () {
    const sql = yield* SqlClient.SqlClient
    
    const statement = yield* sql.prepare(`
      SELECT * FROM users WHERE id = $1
    `)
    
    return yield* statement.execute([id])
  })
```

#### Drizzle Integration

**Type-Safe Query Building:**

```typescript
import { PgDrizzle } from "@effect/sql-drizzle/Pg"
import { Effect } from "effect"
import { eq, and, or, desc } from "drizzle-orm"
import * as schema from "./schema"

// Define Drizzle schema
export const users = pgTable('users', {
  id: uuid('id').primaryKey(),
  name: text('name').notNull(),
  email: text('email').notNull().unique(),
  createdAt: timestamp('created_at').defaultNow()
})

// Use Drizzle with Effect
const fetchUserById = (userId: string) =>
  Effect.gen(function* () {
    const db = yield* PgDrizzle
    
    const result = yield* db
      .select()
      .from(schema.users)
      .where(eq(schema.users.id, userId))
      .limit(1)
    
    if (result.length === 0) {
      return yield* Effect.fail(
        new UserNotFoundError({ userId })
      )
    }
    
    return result[0]
  })

// Complex queries with joins
const getUserWithOrders = (userId: string) =>
  Effect.gen(function* () {
    const db = yield* PgDrizzle
    
    return yield* db
      .select({
        user: schema.users,
        order: schema.orders
      })
      .from(schema.users)
      .leftJoin(
        schema.orders,
        eq(schema.users.id, schema.orders.userId)
      )
      .where(eq(schema.users.id, userId))
      .orderBy(desc(schema.orders.createdAt))
  })

// Insert with returning
const createUser = (userData: NewUser) =>
  Effect.gen(function* () {
    const db = yield* PgDrizzle
    
    const [newUser] = yield* db
      .insert(schema.users)
      .values(userData)
      .returning()
    
    return newUser
  })

// Update with conditions
const updateUserEmail = (userId: string, newEmail: string) =>
  Effect.gen(function* () {
    const db = yield* PgDrizzle
    
    const [updated] = yield* db
      .update(schema.users)
      .set({ email: newEmail })
      .where(eq(schema.users.id, userId))
      .returning()
    
    if (!updated) {
      return yield* Effect.fail(
        new UserNotFoundError({ userId })
      )
    }
    
    return updated
  })
```

#### Model Patterns with @effect/sql

**Model.Class for Database Entities:**

```typescript
import { Model } from "@effect/sql"
import { Schema, DateTime, Option } from "effect"

// Define branded types
const UserId = Schema.String.pipe(Schema.brand("UserId"))
type UserId = Schema.Schema.Type<typeof UserId>

const ProjectId = Schema.String.pipe(Schema.brand("ProjectId"))
type ProjectId = Schema.Schema.Type<typeof ProjectId>

// Define database model
export class User extends Model.Class<User>("User")({
  // Generated fields (auto-populated by database)
  id: Model.Generated(UserId),
  createdAt: Model.DateTimeInsert,
  updatedAt: Model.DateTimeUpdate,
  
  // Required fields
  firstName: Schema.NonEmptyTrimmedString,
  lastName: Schema.NonEmptyTrimmedString,
  email: Schema.String.pipe(
    Schema.pattern(/^[^\s@]+@[^\s@]+\.[^\s@]+$/)
  ),
  
  // Optional fields
  dateOfBirth: Model.FieldOption(Model.Date),
  phoneNumber: Schema.optional(Schema.String),
  projectId: Schema.optional(ProjectId)
}) {}

// Different schemas for different use cases
type UserEntity = typeof User.Type          // Full entity
type UserInsert = typeof User.insert.Type   // For INSERT
type UserUpdate = typeof User.update.Type   // For UPDATE
type UserJson = typeof User.json.Type       // For API responses
type UserJsonCreate = typeof User.jsonCreate.Type  // For API input
type UserJsonUpdate = typeof User.jsonUpdate.Type  // For API update

// Using the model in repository
export class UserRepository extends Effect.Service<UserRepository>()(
  "UserRepository",
  {
    effect: Effect.gen(function* () {
      const sql = yield* SqlClient.SqlClient
      
      return {
        create: (data: UserJsonCreate) =>
          Effect.gen(function* () {
            // Validate and transform input
            const insertData = yield* Schema.decode(User.jsonCreate)(data)
            
            const [user] = yield* sql`
              INSERT INTO users ${sql.insert(insertData)}
              RETURNING *
            `
            
            return yield* Schema.decode(User)(user)
          }),
        
        findById: (id: UserId) =>
          Effect.gen(function* () {
            const result = yield* sql`
              SELECT * FROM users WHERE id = ${id}
            `
            
            if (result.rows.length === 0) {
              return yield* Effect.fail(
                new UserNotFoundError({ userId: id })
              )
            }
            
            return yield* Schema.decode(User)(result.rows[0])
          }),
        
        update: (id: UserId, data: UserJsonUpdate) =>
          Effect.gen(function* () {
            const updateData = yield* Schema.decode(User.jsonUpdate)(data)
            
            const [updated] = yield* sql`
              UPDATE users
              SET ${sql.update(updateData)}
              WHERE id = ${id}
              RETURNING *
            `
            
            if (!updated) {
              return yield* Effect.fail(
                new UserNotFoundError({ userId: id })
              )
            }
            
            return yield* Schema.decode(User)(updated)
          })
      }
    }),
    dependencies: [SqlClient.SqlClient.Default]
  }
) {}
```

```typescript
// Migration service
export class MigrationService extends Effect.Service<MigrationService>()(
  "MigrationService",
  {
    effect: Effect.gen(function* () {
      const sql = yield* SqlClient.SqlClient
      const logger = yield* Logger.Logger
      
      const runMigration = (version: string, sql: string) =>
        Effect.gen(function* () {
          yield* logger.info(`Running migration ${version}`)
          yield* sql.withTransaction(
            Effect.gen(function* () {
              yield* sql.unsafe(migrationSql)
              yield* sql`
                INSERT INTO migrations (version, applied_at)
                VALUES (${version}, NOW())
              `
            })
          )
          yield* logger.info(`Migration ${version} completed`)
        })
      
      return {
        migrate: Effect.gen(function* () {
          // Create migrations table if not exists
          yield* sql`
            CREATE TABLE IF NOT EXISTS migrations (
              version VARCHAR(255) PRIMARY KEY,
              applied_at TIMESTAMP NOT NULL
            )
          `
          
          // Get applied migrations
          const applied = yield* sql`
            SELECT version FROM migrations
          `.pipe(
            Effect.map(result => 
              new Set(result.rows.map(row => row.version))
            )
          )
          
          // Run pending migrations
          for (const migration of pendingMigrations) {
            if (!applied.has(migration.version)) {
              yield* runMigration(migration.version, migration.sql)
            }
          }
        })
      }
    }),
    dependencies: [SqlClient.SqlClient.Default, Logger.Logger.Default]
  }
```

### 17.2 Message Queues

#### Queue Patterns

**Producer Pattern:**

```typescript
import { Effect, Queue } from "effect"

// Define message schema
const OrderMessage = Schema.Struct({
  orderId: Schema.String,
  customerId: Schema.String,
  items: Schema.Array(Schema.Struct({
    productId: Schema.String,
    quantity: Schema.Number
  })),
  timestamp: Schema.DateTimeUtc
})

type OrderMessage = Schema.Schema.Type<typeof OrderMessage>

// Producer service
export class OrderProducer extends Effect.Service<OrderProducer>()(
  "OrderProducer",
  {
    effect: Effect.gen(function* () {
      const queue = yield* Queue.unbounded<OrderMessage>()
      const logger = yield* Logger.Logger
      
      return {
        publish: (message: OrderMessage) =>
          Effect.gen(function* () {
            // Validate message
            const validated = yield* Schema.decode(OrderMessage)(message)
            
            yield* logger.info("Publishing order message", {
              orderId: validated.orderId
            })
            
            yield* Queue.offer(queue, validated)
            
            yield* logger.debug("Order message published", {
              orderId: validated.orderId,
              queueSize: yield* Queue.size(queue)
            })
          }).pipe(
            Effect.withSpan("order.publish", {
              attributes: { "order.id": message.orderId }
            })
          ),
        
        getQueue: Effect.succeed(queue)
      }
    }),
    dependencies: [Logger.Logger.Default]
  }
) {}
```

**Consumer Pattern:**

```typescript
// Consumer with error handling
export class OrderConsumer extends Effect.Service<OrderConsumer>()(
  "OrderConsumer",
  {
    effect: Effect.gen(function* () {
      const producer = yield* OrderProducer
      const orderService = yield* OrderService
      const logger = yield* Logger.Logger
      const queue = yield* producer.getQueue
      
      const processMessage = (message: OrderMessage) =>
        Effect.gen(function* () {
          yield* logger.info("Processing order", {
            orderId: message.orderId
          })
          
          yield* orderService.processOrder(message)
          
          yield* logger.info("Order processed successfully", {
            orderId: message.orderId
          })
        }).pipe(
          Effect.retry({
            times: 3,
            schedule: Schedule.exponential("100 millis")
          }),
          Effect.catchAll((error) =>
            Effect.gen(function* () {
              yield* logger.error("Failed to process order", {
                orderId: message.orderId,
                error
              })
              
              // Send to dead letter queue
              yield* sendToDeadLetter(message, error)
            })
          ),
          Effect.withSpan("order.process", {
            attributes: { "order.id": message.orderId }
          })
        )
      
      const consume = Effect.gen(function* () {
        while (true) {
          const message = yield* Queue.take(queue)
          yield* processMessage(message)
        }
      }).pipe(
        Effect.catchTag("QueueShutdown", () => 
          logger.info("Consumer shutting down")
        )
      )
      
      return { consume }
    }),
    dependencies: [
      OrderProducer.Default,
      OrderService.Default,
      Logger.Logger.Default
    ]
  }
) {}
```

**At-Least-Once Delivery:**

```typescript
// Idempotent message processing
export class IdempotentOrderProcessor extends Effect.Service<IdempotentOrderProcessor>()(
  "IdempotentOrderProcessor",
  {
    effect: Effect.gen(function* () {
      const sql = yield* SqlClient.SqlClient
      const logger = yield* Logger.Logger
      
      return {
        process: (message: OrderMessage) =>
          Effect.gen(function* () {
            // Check if message already processed
            const existing = yield* sql`
              SELECT id FROM processed_messages
              WHERE message_id = ${message.orderId}
            `
            
            if (existing.rows.length > 0) {
              yield* logger.info("Message already processed", {
                orderId: message.orderId
              })
              return // Idempotent - skip processing
            }
            
            // Process within transaction
            yield* sql.withTransaction(
              Effect.gen(function* () {
                // Process order
                yield* processOrder(message)
                
                // Mark as processed
                yield* sql`
                  INSERT INTO processed_messages (
                    message_id,
                    processed_at
                  ) VALUES (
                    ${message.orderId},
                    NOW()
                  )
                `
              })
            )
            
            yield* logger.info("Order processed", {
              orderId: message.orderId
            })
          })
      }
    }),
    dependencies: [SqlClient.SqlClient.Default, Logger.Logger.Default]
  }
) {}
```

#### Circuit Breaker Pattern

```typescript
import { Effect, Ref } from "effect"

type CircuitState = 
  | { _tag: "Closed" }
  | { _tag: "Open"; openedAt: number }
  | { _tag: "HalfOpen" }

// Circuit breaker service
export class CircuitBreaker extends Effect.Service<CircuitBreaker>()(
  "CircuitBreaker",
  {
    effect: Effect.gen(function* () {
      const config = yield* Config.all({
        failureThreshold: Config.number("CIRCUIT_FAILURE_THRESHOLD")
          .pipe(Config.withDefault(5)),
        resetTimeout: Config.duration("CIRCUIT_RESET_TIMEOUT")
          .pipe(Config.withDefault("30 seconds"))
      })
      
      const state = yield* Ref.make<CircuitState>({ _tag: "Closed" })
      const failureCount = yield* Ref.make(0)
      const logger = yield* Logger.Logger
      
      const shouldAllowRequest = Effect.gen(function* () {
        const currentState = yield* Ref.get(state)
        
        switch (currentState._tag) {
          case "Closed":
            return true
          
          case "Open": {
            const elapsed = Date.now() - currentState.openedAt
            const timeoutMs = Duration.toMillis(config.resetTimeout)
            
            if (elapsed >= timeoutMs) {
              yield* logger.info("Circuit transitioning to half-open")
              yield* Ref.set(state, { _tag: "HalfOpen" })
              return true
            }
            
            return false
          }
          
          case "HalfOpen":
            return true
        }
      })
      
      const recordSuccess = Effect.gen(function* () {
        const currentState = yield* Ref.get(state)
        
        if (currentState._tag === "HalfOpen") {
          yield* logger.info("Circuit closing after successful request")
          yield* Ref.set(state, { _tag: "Closed" })
          yield* Ref.set(failureCount, 0)
        } else if (currentState._tag === "Closed") {
          yield* Ref.set(failureCount, 0)
        }
      })
      
      const recordFailure = Effect.gen(function* () {
        const currentState = yield* Ref.get(state)
        const count = yield* Ref.updateAndGet(
          failureCount,
          (n) => n + 1
        )
        
        if (currentState._tag === "HalfOpen") {
          yield* logger.warn("Circuit opening after failed half-open attempt")
          yield* Ref.set(state, { 
            _tag: "Open",
            openedAt: Date.now()
          })
        } else if (
          currentState._tag === "Closed" &&
          count >= config.failureThreshold
        ) {
          yield* logger.warn("Circuit opening due to failure threshold")
          yield* Ref.set(state, {
            _tag: "Open",
            openedAt: Date.now()
          })
        }
      })
      
      return {
        execute: <A, E, R>(
          effect: Effect.Effect<A, E, R>
        ): Effect.Effect<A, E | CircuitOpenError, R> =>
          Effect.gen(function* () {
            const allowed = yield* shouldAllowRequest
            
            if (!allowed) {
              return yield* Effect.fail(new CircuitOpenError())
            }
            
            const result = yield* Effect.either(effect)
            
            if (result._tag === "Right") {
              yield* recordSuccess
              return result.right
            } else {
              yield* recordFailure
              return yield* Effect.fail(result.left)
            }
          })
      }
    }),
    dependencies: [Logger.Logger.Default]
  }
) {}

// Usage
const fetchWithCircuitBreaker = (userId: string) =>

    const circuitBreaker = yield* CircuitBreaker
    const apiClient = yield* ExternalApiClient

      apiClient.getUser(userId)
    )
  }).pipe(
    Effect.catchTag("CircuitOpenError", () =>

    )
  )
```

```typescript
// Token bucket rate limiter
export class RateLimiter extends Effect.Service<RateLimiter>()(
  "RateLimiter",
  {
    effect: Effect.gen(function* () {
      const config = yield* Config.all({
        requestsPerSecond: Config.number("RATE_LIMIT_RPS")
          .pipe(Config.withDefault(10))
      })
      
      const tokens = yield* Ref.make(config.requestsPerSecond)
      const lastRefill = yield* Ref.make(Date.now())
      
      const refillTokens = Effect.gen(function* () {
        const now = Date.now()
        const last = yield* Ref.get(lastRefill)
        const elapsed = (now - last) / 1000
        
        if (elapsed >= 1) {
          yield* Ref.set(tokens, config.requestsPerSecond)
          yield* Ref.set(lastRefill, now)
        }
      })
      
      const acquire = Effect.gen(function* () {
        yield* refillTokens
        
        const current = yield* Ref.get(tokens)
        
        if (current > 0) {
          yield* Ref.update(tokens, (n) => n - 1)
          return true
        }
        
        return false
      })
      
      return {
        execute: <A, E, R>(
          effect: Effect.Effect<A, E, R>
        ): Effect.Effect<A, E | RateLimitError, R> =>
          Effect.gen(function* () {
            const allowed = yield* acquire
            
            if (!allowed) {
              return yield* Effect.fail(new RateLimitError())
            }
            
            return yield* effect
          })
      }
    })
  }
) {}
```

```typescript
import { Effect, Layer, ManagedRuntime } from "effect"
import { createContext, useContext, useEffect, useState } from "react"

// Create runtime context
const RuntimeContext = createContext<ManagedRuntime.ManagedRuntime<AppServices>>(
  null as any
)

// Provider component
export function EffectProvider({ children }: { children: React.ReactNode }) {
  const [runtime] = useState(() =>
    ManagedRuntime.make(AppLayer)
  )
  
  useEffect(() => {
    return () => {
      ManagedRuntime.dispose(runtime)
    }
  }, [runtime])
  
  return (
    <RuntimeContext.Provider value={runtime}>
      {children}
    </RuntimeContext.Provider>
  )
}

// Custom hooks
export function useRuntime() {
  return useContext(RuntimeContext)
}

export function useEffect<A, E>(
  effect: Effect.Effect<A, E, AppServices>,
  deps: React.DependencyList
) {
  const runtime = useRuntime()
  const [data, setData] = useState<A | null>(null)
  const [error, setError] = useState<E | null>(null)
  const [loading, setLoading] = useState(true)
  
  React.useEffect(() => {
    setLoading(true)
    setError(null)
    
    const fiber = ManagedRuntime.runFork(runtime)(effect)
    
    Effect.runPromise(fiber).then(
      (result) => {
        setData(result)
        setLoading(false)
      },
      (error) => {
        setError(error)
        setLoading(false)
      }
    )
    
    return () => {
      ManagedRuntime.interrupt(fiber)
    }
  }, deps)
  
  return { data, error, loading }
}

// Usage in components
function UserProfile({ userId }: { userId: string }) {
  const { data, error, loading } = useEffect(
    Effect.gen(function* () {
      const userService = yield* UserService

    }),
    [userId]
  )
  
  if (loading) return <div>Loading...</div>
  if (error) return <div>Error: {error.message}</div>
  if (!data) return null
  
  return <div>{data.name}</div>
}
```

```typescript
      return newUser

        loading = false
        return newUser

        error = err
        loading = false
        throw err
      }
    )

    error = null

        return updated
      },
      (err) => {
        error = err
        loading = false
        throw err
      }
    )
  }

    get users() { return users },
    get loading() { return loading },
    get error() { return error },
```

### Best Practices Summary

**Database Integration:**
1. Use `@effect/sql` for type-safe database operations
2. Wrap all queries in proper error handling
3. Use transactions for multi-step operations
4. Implement connection pooling for scalability
5. Use `Model.Class` for database entities with automatic validation

**Message Queues:**
1. Validate messages with Schema at queue boundaries
2. Implement idempotent message processing
3. Use dead letter queues for failed messages
4. Implement proper retry logic with exponential backoff
5. Monitor queue depth and processing latency

**External Services:**
1. Wrap external APIs in Effect services for testability
2. Implement circuit breakers for fault tolerance
3. Use rate limiting to prevent overwhelming services
4. Add proper retry logic with backoff strategies
5. Implement bulkhead pattern to isolate failures

**Frontend Integration:**
1. Use `ManagedRuntime` for React integration
2. Leverage Svelte 5 runes for reactive state with Effect
3. Create composable hooks/stores for reusable logic
4. Handle loading and error states consistently
5. Provide layers at application root level
